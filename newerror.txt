    similarity_dict[ctrl_right][ctrl_left] = sim_score
    
    # Store for easy access

    similar_controls[ctrl_right].append((ctrl_left, sim_score))

def get_llm_expansion(
    regulation_id: str,
    regulatory_text: str,
    llm_controls: Set[str],
    all_controls: Set[str],
    control_descriptions: dict,
    model: str = "gpt-4"
) -> List[str]:
    """
    Use LLM to find additional controls that should map to this regulation.
    """
    
    # Create control catalog with descriptions
    control_catalog = "AVAILABLE CONTROLS:\n\n"
    for ctrl in sorted(all_controls):
        desc = control_descriptions.get(ctrl, "No description")
        control_catalog += f"{ctrl}: {desc}\n"
    
    system_prompt = f"""You are a compliance expert who maps regulatory requirements to security controls.

Below is the complete catalog of available controls:

{control_catalog}

Your task: Analyze regulatory requirements and identify ALL relevant controls from this catalog.

RULES:
1. Consider ALL controls that relate to the requirement in any way
2. Include direct implementations and supporting controls
3. Think broadly about related controls
4. Return ONLY valid control IDs from the catalog
5. Return as comma-separated list with no other text
"""

    user_prompt = f"""Regulation: {regulatory_text}

LLM initially identified: {','.join(sorted(llm_controls))}

Find ALL relevant controls (including the ones above). Think carefully about:
- Direct implementations
- Supporting controls
- Related controls
- Preventive controls
- Detective controls
- Corrective controls

Return complete comma-separated list of ALL relevant control IDs.
Return ONLY the control IDs, nothing else.
"""

    try:
        response = openai.ChatCompletion.create(
            model=model,
            messages=[
                {"role": "system", "content": system_prompt},
                {"role": "user", "content": user_prompt}
            ],
            temperature=0.3,
            max_tokens=2000
        )
        
        response_text = response.choices[0].message.content.strip()
        
        # Parse control IDs
        controls = [c.strip() for c in response_text.split(',')]
        controls = [c for c in controls if c and c in all_controls]
        
        return controls
    
    except Exception as e:
        print(f"  Error in LLM expansion for {regulation_id}: {str(e)}")
        return list(llm_controls)

def expand_by_neighbors(
    controls: Set[str],
    similarity_dict: dict,
    control_neighbors: dict,
    similarity_threshold: float = 0.65,
    max_neighbors_per_control: int = 3
) -> Set[str]:
    """
    For each control, add its similar neighbors.
    """
    
    expanded = set(controls)
    
    for ctrl in controls:
        if ctrl in control_neighbors:
            # Add top similar neighbors
            for neighbor, sim_score in control_neighbors[ctrl][:max_neighbors_per_control]:
                if sim_score >= similarity_threshold:
                    expanded.add(neighbor)
    
    return expanded

        controls=llm_set,

        control_neighbors=control_neighbors,

        max_neighbors_per_control=3

def expand_by_transitivity(
    controls: Set[str],
    similarity_dict: dict,
    similarity_threshold: float = 0.70,
    max_hops: int = 1
) -> Set[str]:
    """
    Find controls that are similar to controls similar to our original controls.
    This creates a network effect to find related controls.
    """
    
    expanded = set(controls)
    current_level = set(controls)
    
    for hop in range(max_hops):
        next_level = set()
        
        # For each control at current level
        for ctrl in current_level:
            if ctrl in similarity_dict:
                # Find all similar controls
                for similar_ctrl, sim_score in similarity_dict[ctrl].items():
                    if sim_score >= similarity_threshold and similar_ctrl not in expanded:
                        next_level.add(similar_ctrl)
                        expanded.add(similar_ctrl)
        
        current_level = next_level
        if not next_level:
            break
    
    return expanded

def discover_by_keywords(
    regulatory_text: str,
    control_descriptions: dict,
    current_controls: Set[str],
    top_k: int = 5
) -> Set[str]:
    """
    Find controls whose descriptions match keywords in regulatory text.
    """
    
    discovered = set(current_controls)
    
    # Get keywords from regulatory text (simple approach: split and filter)
    keywords = set(regulatory_text.lower().split())
    keywords = {kw for kw in keywords if len(kw) > 3}  # Filter short words
    
    # Score each control based on keyword overlap
    control_scores = {}
    
    for ctrl_id, description in control_descriptions.items():
        if ctrl_id in current_controls:
            continue
        
        desc_lower = description.lower()
        score = sum(1 for kw in keywords if kw in desc_lower)
        
        if score > 0:
            control_scores[ctrl_id] = score
    
    # Get top K controls by keyword match
    if control_scores:
        top_controls = sorted(control_scores.items(), key=lambda x: x[1], reverse=True)[:top_k]
        for ctrl_id, score in top_controls:
            discovered.add(ctrl_id)
    
    return discovered

        regulatory_text=regulatory_text,
        control_descriptions=control_descriptions,
        current_controls=llm_set,
        top_k=5

def combine_all_expansions(
    llm_controls: Set[str],
    llm_expanded: List[str],
    similarity_expanded: List[str],
    transitive_expanded: List[str],
    keyword_discovered: List[str],
    max_total: int = 20
) -> Set[str]:
    """
    Intelligently combine all expansion methods.
    Priority: LLM > Transitive > Similarity > Keyword
    """
    
    combined = set(llm_expanded)  # LLM expansion has highest priority
    
    # Add transitive (second priority)
    for ctrl in transitive_expanded:
        if len(combined) < max_total:
            combined.add(ctrl)
    
    # Add similarity
    for ctrl in similarity_expanded:
        if len(combined) < max_total:
            combined.add(ctrl)
    
    # Add keyword discovered
    for ctrl in keyword_discovered:
        if len(combined) < max_total:
            combined.add(ctrl)
    
    # Ensure original LLM controls are included
    combined.update(llm_controls)
    
    # If too many, keep top by frequency in all sources
    if len(combined) > max_total:
        # Score by how many sources suggested it
        control_scores = defaultdict(int)
        
        for ctrl in llm_expanded:
            control_scores[ctrl] += 3  # LLM gets highest weight
        for ctrl in transitive_expanded:
            control_scores[ctrl] += 2
        for ctrl in similarity_expanded:
            control_scores[ctrl] += 1
        for ctrl in keyword_discovered:
            control_scores[ctrl] += 1
        
        # Keep original LLM controls and top scored
        top_controls = sorted(control_scores.items(), key=lambda x: x[1], reverse=True)
        combined = set(llm_controls)
        
        for ctrl, score in top_controls:
            if ctrl not in combined and len(combined) < max_total:
                combined.add(ctrl)
    
    return combined


print("\nCombining all expansion strategies...")
