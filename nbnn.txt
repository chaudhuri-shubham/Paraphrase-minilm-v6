import pandas as pd
import os
import json
from openai import OpenAI
from dotenv import load_dotenv

# --- 1. Configuration ---
load_dotenv()
client = OpenAI(api_key=os.getenv("OPENAI_API_KEY"))

# --- 2. Prompt Template Function ---
def create_taxonomy_prompt(reg_text, level1_inputs, reference_df):
    """
    Filters the reference dataframe for relevant Level 3 options 
    and builds the final classification prompt.
    """
    # Filter reference data for the specific Level 1 categories provided
    filtered_df = reference_df[reference_df['Level 1'].isin(level1_inputs)]
    
    # Build the Reference Schema string
    schema_list = []
    for _, row in filtered_df.iterrows():
        schema_list.append(f"- Level 3: {row['Level 3 Text']} | Definition: {row['Definition']}")
    
    schema_str = "\n".join(schema_list)

    prompt = f"""
Role: You are a Regulatory Compliance Expert and Taxonomy Specialist.

Task: Analyze the Regulatory Text and map it to the most relevant Level 3 Categories from the Reference Schema below.

Regulatory Text: "{reg_text}"
Target Level 1 Context: {', '.join(level1_inputs)}

### Reference Schema (Filtered for Level 1 context) ###
{schema_str}

### Instructions ###
1. Only select categories listed in the Reference Schema above.
2. You may select a single category, multiple categories, or "No Match" if none apply.
3. Compare the intent of the Regulatory Text against the Definitions provided.

Output ONLY a JSON object with this structure:
{{
  "selected_level_3": ["Category Name 1", "Category Name 2"],
  "reasoning": "Brief explanation of the match based on definitions"
}}
"""
    return prompt

# --- 3. Classification Core ---
def classify_regulatory_row(row, reference_df, text_col, l1_col):
    """
    Handles a single row of the input dataframe.
    """
    reg_text = row[text_col]
    # Handle both single string Level 1 or a list of Level 1s
    l1_inputs = row[l1_col] if isinstance(row[l1_col], list) else [row[l1_col]]
    
    prompt = create_taxonomy_prompt(reg_text, l1_inputs, reference_df)
    
    try:
        response = client.chat.completions.create(
            model="gpt-4o",
            messages=[{"role": "user", "content": prompt}],
            response_format={"type": "json_object"},
            temperature=0
        )
        return json.loads(response.choices[0].message.content)
    except Exception as e:
        return {"selected_level_3": ["Error"], "reasoning": str(e)}

# --- 4. Main Pipeline ---
def run_classification_pipeline(input_df, reference_df, text_col, l1_col):
    print(f"Starting classification for {len(input_df)} records...")
    
    # Apply logic to each row
    results = input_df.apply(lambda row: classify_regulatory_row(row, reference_df, text_col, l1_col), axis=1)
    
    # Normalize JSON into columns
    results_df = pd.json_normalize(results)
    
    # Merge with original input_df to preserve all columns
    final_df = pd.concat([input_df.reset_index(drop=True), results_df], axis=1)
    
    return final_df

import os
import fitz                            # PyMuPDF
import math
import pickle
import numpy as np
from tqdm import tqdm
from sentence_transformers import SentenceTransformer
from typing import List, Dict, Tuple

# Try to import faiss; if not present we will fallback
try:
    import faiss
    _FAISS_AVAILABLE = True
except Exception:
    _FAISS_AVAILABLE = False
    faiss = None

# ---------------------------
# CONFIG
# ---------------------------
PDF_FOLDER = "pdfs"               # folder containing PDF files
EMBEDDINGS_PATH = "pdf_embeddings.pkl"   # to save embeddings + metadata
FAISS_INDEX_PATH = "faiss.index" # optional path for faiss index
MODEL_NAME = "all-MiniLM-L6-v2"  # MiniLM model
CHUNK_SIZE = 800                 # characters per chunk (tune)
CHUNK_OVERLAP = 200              # overlap characters between chunks
EMBED_BATCH_SIZE = 64
DEVICE = "cuda" if __import__("torch").cuda.is_available() else "cpu"

# ---------------------------
#  UTIL: PDF TEXT EXTRACTION
# ---------------------------
def extract_text_from_pdf(path: str) -> str:
    """Extracts and concatenates text from all pages of a PDF using PyMuPDF."""
    text_pages = []
    with fitz.open(path) as doc:
        for page in doc:
            txt = page.get_text("text")  # "text" extracts readable text
            if txt:
                text_pages.append(txt)
    return "\n".join(text_pages)

# ---------------------------
#  UTIL: TEXT CHUNKING
# ---------------------------
def chunk_text(text: str, chunk_size: int = CHUNK_SIZE, overlap: int = CHUNK_OVERLAP) -> List[str]:
    """
    Splits text into chunks of approximately chunk_size characters with given overlap.
    Uses a simple sliding-window on characters; can be replaced with sentence-based splitting.
    """
    if not text:
        return []
    text = text.replace("\r\n", "\n")
    chunks = []
    start = 0
    text_len = len(text)
    while start < text_len:
        end = start + chunk_size
        chunk = text[start:end]
        chunks.append(chunk.strip())
        # move forward with overlap
        start = end - overlap
        if start < 0:
            start = 0
    return [c for c in chunks if c]

# ---------------------------
#  INDEX CREATION
# ---------------------------
def index_pdfs(pdf_folder: str = PDF_FOLDER,
               model_name: str = MODEL_NAME,
               embeddings_path: str = EMBEDDINGS_PATH,
               use_faiss: bool = _FAISS_AVAILABLE):
    """
    Walks the pdf_folder, extracts chunks from PDFs, computes embeddings, and saves them.
    Returns metadata dict with: embeddings (ndarray), metadatas (list of dicts)
    """
    model = SentenceTransformer(model_name, device=DEVICE)
    metadata = []   # list of {file, chunk_id, text}
    embeddings_list = []

    # Gather all pdf files
    pdf_files = [os.path.join(pdf_folder, f) for f in os.listdir(pdf_folder)
                 if f.lower().endswith(".pdf")]

    print(f"Found {len(pdf_files)} PDF files.")

    for pdf_path in tqdm(pdf_files, desc="PDFs"):
        try:
            raw_text = extract_text_from_pdf(pdf_path)
        except Exception as e:
            print(f"Failed to extract {pdf_path}: {e}")
            continue

        chunks = chunk_text(raw_text, CHUNK_SIZE, CHUNK_OVERLAP)
        if not chunks:
            continue

        # compute embeddings in batches for this pdf
        for i in range(0, len(chunks), EMBED_BATCH_SIZE):
            batch = chunks[i:i+EMBED_BATCH_SIZE]
            embs = model.encode(batch, show_progress_bar=False, convert_to_numpy=True, batch_size=EMBED_BATCH_SIZE)
            # append embeddings and metadata in matching order
            for j, emb in enumerate(embs):
                chunk_id = i + j
                metadata.append({
                    "file": os.path.basename(pdf_path),
                    "chunk_id": chunk_id,
                    "text": batch[j]
                })
                embeddings_list.append(emb.astype(np.float32))

    if len(embeddings_list) == 0:
        raise RuntimeError("No text chunks were extracted; check PDF content.")

    embeddings = np.vstack(embeddings_list)  # shape (N, d)
    print(f"Produced {embeddings.shape[0]} embeddings of dimension {embeddings.shape[1]}")

    # Save embeddings + metadata to disk (pickle)
    with open(embeddings_path, "wb") as f:
        pickle.dump({
            "embeddings": embeddings,
            "metadata": metadata,
            "model_name": model_name,
            "chunk_size": CHUNK_SIZE,
            "chunk_overlap": CHUNK_OVERLAP
        }, f)
    print(f"Embeddings & metadata saved to {embeddings_path}")

    # Optionally build faiss index
    if use_faiss:
        d = embeddings.shape[1]
        index = faiss.IndexFlatIP(d)  # inner product index for cosine when vectors normalized
        # normalize embeddings to unit length for cosine similarity with inner product
        faiss.normalize_L2(embeddings)
        index.add(embeddings)
        faiss.write_index(index, FAISS_INDEX_PATH)
        print(f"FAISS index built and saved to {FAISS_INDEX_PATH}")

    return {"embeddings": embeddings, "metadata": metadata}

# ---------------------------
#  LOADING INDEX
# ---------------------------
def load_index(embeddings_path: str = EMBEDDINGS_PATH):
    with open(embeddings_path, "rb") as f:
        data = pickle.load(f)
    embeddings = data["embeddings"]
    metadata = data["metadata"]
    return embeddings, metadata

# ---------------------------
#  SEARCH FUNCTION
# ---------------------------
def semantic_search(query: str,
                    top_k: int = 5,
                    embeddings_path: str = EMBEDDINGS_PATH,
                    use_faiss: bool = _FAISS_AVAILABLE) -> List[Dict]:
    """
    Returns top_k results for query as a list of dicts {score, file, chunk_id, text}
    """
    # load model & index
    model = SentenceTransformer(MODEL_NAME, device=DEVICE)
    q_emb = model.encode([query], convert_to_numpy=True)[0].astype(np.float32)

    embeddings, metadata = load_index(embeddings_path)

    results = []
    if use_faiss:
        # load faiss index
        index = faiss.read_index(FAISS_INDEX_PATH)
        # faiss expects normalized vectors if using inner product as cosine
        faiss.normalize_L2(q_emb.reshape(1, -1))
        D, I = index.search(q_emb.reshape(1, -1), top_k)
        # D are inner products (cosine) after normalization
        for score, idx in zip(D[0], I[0]):
            if idx < 0:
                continue
            meta = metadata[idx]
            results.append({
                "score": float(score),
                "file": meta["file"],
                "chunk_id": meta["chunk_id"],
                "text": meta["text"]
            })
    else:
        # Brute-force cosine similarity fallback
        # normalize embeddings and q_emb
        def l2_norm(a):
            return a / np.linalg.norm(a, axis=1, keepdims=True)
        E = embeddings.copy()
        # handle degenerate zeros
        E_norm = E / np.clip(np.linalg.norm(E, axis=1, keepdims=True), a_min=1e-10, a_max=None)
        q_norm = q_emb / max(1e-10, np.linalg.norm(q_emb))
        scores = (E_norm @ q_norm).astype(np.float32)
        top_idx = np.argsort(-scores)[:top_k]
        for idx in top_idx:
            results.append({
                "score": float(scores[idx]),
                "file": metadata[idx]["file"],
                "chunk_id": metadata[idx]["chunk_id"],
                "text": metadata[idx]["text"]
            })

    return results
