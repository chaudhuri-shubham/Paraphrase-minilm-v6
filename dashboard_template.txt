   than the browser processes in one frame. Cosmetic only; filtered here so
   it doesn't clutter the console. */
(function () {
  function isResizeObserverNoise(msg) {
    return typeof msg === 'string' && msg.indexOf('ResizeObserver loop') !== -1;
  }
  window.addEventListener('error', function (e) {
    if (isResizeObserverNoise(e.message)) { e.stopImmediatePropagation(); e.preventDefault(); return false; }
  }, true);
  const originalConsoleError = console.error;
  console.error = function (...args) {
    if (args.length && isResizeObserverNoise(String(args[0]))) return;
    originalConsoleError.apply(console, args);
  };
})();

<script>
/* =========================================================================
   1. DATA
   In production, swap GRAPH_DATA for a fetch() to your backend / API, or
   render this JSON server-side from build_dashboard.py / grc_pipeline.py.
   The catalog contains EVERY node/edge (risk, issue, control, gap) up front;
   the UI reveals subsets of it progressively as the user clicks through.
   ========================================================================= */
const GRAPH_DATA = __GRAPH_DATA_JSON__;

/* =========================================================================
   2. STYLE MAP per node group
   ========================================================================= */
const STYLE = {
  risk:      { shape:'diamond', color:{background:'#A78BFA', border:'#C4B5FD'}, size:24, font:{color:'#E7ECF5', size:13, face:'Inter', bold:true} },
  issue:     { shape:'dot',     color:{background:'#4F9CF9', border:'#8FC1FF'}, size:16, font:{color:'#C9D6EC', size:11.5, face:'Inter'} },
  issue_gap: { shape:'dot',     color:{background:'#F5A524', border:'#FBD097'}, size:16, font:{color:'#C9D6EC', size:11.5, face:'Inter'} },
  control:   { shape:'box',     color:{background:'#173B2E', border:'#34D3A3'}, size:14, font:{color:'#C9D6EC', size:11, face:'Inter'} },
  gap:       { shape:'triangle',color:{background:'#3A1613', border:'#F1554C'}, size:13, font:{color:'#F1554C', size:10.5, face:'Inter'} },
  theme:     { shape:'star',    color:{background:'#0E4A55', border:'#22D3EE'}, size:20, font:{color:'#22D3EE', size:12, face:'Inter', bold:true} },
};

function makeVisNode(n){
  const s = STYLE[n.group];
  const showIdAsLabel = (n.group === 'issue' || n.group === 'issue_gap' || n.group === 'control');
  const base = {
    id: n.id,
    label: showIdAsLabel ? n.id : n.label,
    title: n.label,
    _raw: n,
    borderWidth: 2,
  };
  if (n.group === 'gap'){ base.shapeProperties = { borderDashes: [4,3] }; }
  return Object.assign(base, s);
}

function themeNodeId(themeName){ return 'THEME::' + themeName; }

function makeThemeVisNode(themeName){
  return Object.assign({
    id: themeNodeId(themeName),
    label: themeName,
    title: themeName + ' (theme)',
    _raw: { id: themeNodeId(themeName), group:'theme', label:themeName, details:{ type:'Theme', name:themeName } },
    borderWidth: 2,
  }, STYLE.theme);
}

function initApp(){

const nodesDS = new vis.DataSet([]);
const edgesDS = new vis.DataSet([]);

/* Base network at load time:
   - MAPPED issues appear already connected to their Risk Taxonomy and Control
     (this is the "already in the network" state).
   - GAP issues appear as isolated nodes with no edges at all - they are
     "outside the network" until the user tags a theme, expands it to find
     similar issues, and submits a recommended control (see section 5). */
GRAPH_DATA.nodes
  .filter(n => n.group === 'issue' || n.group === 'issue_gap')
  .forEach(n => nodesDS.add(makeVisNode(n)));

/* =========================================================================
   3. NETWORK
   ========================================================================= */
const container = document.getElementById('graph');
const network = new vis.Network(container, { nodes: nodesDS, edges: edgesDS }, {
  layout:{ improvedLayout:true },
  interaction:{ hover:true, tooltipDelay:120 },
  physics:{
    solver:'forceAtlas2Based',
    forceAtlas2Based:{ gravitationalConstant:-70, springLength:120, springConstant:0.06, avoidOverlap:0.6 },
    stabilization:{ iterations:150 },
  },
  nodes:{ borderWidthSelected:3 },
  edges:{ arrows:{ to:{enabled:true, scaleFactor:0.5} } },
});

/* =========================================================================
   4. RUNTIME STATE
   ========================================================================= */
const selectedThemesByIssue = {};   // issueId -> Set(themeName)
const resolvedGaps = new Set();     // issueIds that had a recommendation submitted
let dynamicResolvedCount = 0;const totalRisks   = GRAPH_DATA.nodes.filter(n => n.group === 'risk').length;
const totalIssues  = GRAPH_DATA.nodes.filter(n => n.group === 'issue' || n.group === 'issue_gap').length;
const baseMapped   = GRAPH_DATA.nodes.filter(n => n.group === 'control').length;
const baseGaps     = GRAPH_DATA.nodes.filter(n => n.group === 'gap').length;

function refreshStats(){
  document.getElementById('statRisks').textContent  = totalRisks;
  document.getElementById('statIssues').textContent = totalIssues;
  document.getElementById('statMapped').textContent = baseMapped + dynamicResolvedCount;
  document.getElementById('statGaps').textContent   = Math.max(0, baseGaps - dynamicResolvedCount);
}
refreshStats();

/* =========================================================================
   5. GRAPH-BUILDING HELPERS
   ========================================================================= */
function ensureCatalogNode(id){
  if (nodesDS.get(id)) return;
  const raw = GRAPH_DATA.nodes.find(n => n.id === id);
  if (raw) nodesDS.add(makeVisNode(raw));
}

function ensureEdge(fromId, toId, opts){
  const id = fromId + '__' + toId;
  if (edgesDS.get(id)) return;
  edgesDS.add(Object.assign({
    id, from: fromId, to: toId,
    color:{ color:'rgba(143,160,192,0.35)', highlight:'#4F9CF9' },
    width: 1,
    smooth:{ type:'continuous', roundness:0.4 },
  }, opts || {}));
}

function issueRiskEdge(issueId){ return GRAPH_DATA.edges.find(e => e.to === issueId && e.label === 'categorized under'); }
function issueControlEdge(issueId){ return GRAPH_DATA.edges.find(e => e.from === issueId && e.label === 'mitigated by'); }
function issueGapEdge(issueId){ return GRAPH_DATA.edges.find(e => e.from === issueId && e.label === 'control gap'); }

function issuesForTheme(themeName){
  return GRAPH_DATA.nodes.filter(n =>
    (n.group === 'issue' || n.group === 'issue_gap') &&
    n.details && (n.details.theme === themeName || (n.details.possible_themes || []).includes(themeName))
  );
}

function revealIssueRiskAndControl(issueId){
  const rEdge = issueRiskEdge(issueId);
  if (rEdge){ ensureCatalogNode(rEdge.from); ensureEdge(rEdge.from, rEdge.to, { color:{color:'rgba(167,139,250,0.4)', highlight:'#A78BFA'} }); }

  const cEdge = issueControlEdge(issueId);
  if (cEdge){
    ensureCatalogNode(cEdge.to);
    ensureEdge(cEdge.from, cEdge.to, { color:{color:'rgba(52,211,163,0.45)', highlight:'#34D3A3'} });
    return;
  }
  const gEdge = issueGapEdge(issueId);
  if (gEdge && !resolvedGaps.has(issueId)){
    ensureCatalogNode(gEdge.to);
    ensureEdge(gEdge.from, gEdge.to, { dashes:true, color:{color:'rgba(241,85,76,0.55)', highlight:'#F1554C'} });
  }
}

function toggleThemeForIssue(issueId, themeName, chipEl){
  const tId = themeNodeId(themeName);
  if (!selectedThemesByIssue[issueId]) selectedThemesByIssue[issueId] = new Set();
  const set = selectedThemesByIssue[issueId];
  const edgeId = issueId + '__' + tId;

  if (set.has(themeName)){
    set.delete(themeName);
    edgesDS.remove(edgeId);
    const stillUsed = edgesDS.get({ filter: e => e.from === tId || e.to === tId }).length > 0;
    if (!stillUsed) nodesDS.remove(tId);
    if (chipEl) chipEl.classList.remove('selected');
  } else {
    set.add(themeName);
    if (!nodesDS.get(tId)) nodesDS.add(makeThemeVisNode(themeName));
    ensureEdge(issueId, tId, { color:{color:'rgba(34,211,238,0.45)', highlight:'#22D3EE'} });
    if (chipEl) chipEl.classList.add('selected');
  }
}

function expandTheme(themeName){
  const tId = themeNodeId(themeName);
  const similar = issuesForTheme(themeName);
  similar.forEach(issueNode => {
    ensureCatalogNode(issueNode.id);
    ensureEdge(tId, issueNode.id, { color:{color:'rgba(34,211,238,0.35)', highlight:'#22D3EE'} });
    revealIssueRiskAndControl(issueNode.id);
  });
}

function computeRecommendedControls(issueId, themeSet){
  const recs = new Map(); // controlId -> {controlNode, viaIssues:[]}
  themeSet.forEach(themeName => {
    issuesForTheme(themeName).forEach(sim => {
      if (sim.id === issueId || sim.group !== 'issue') return;
      const cEdge = issueControlEdge(sim.id);
      if (!cEdge) return;
      const ctlNode = GRAPH_DATA.nodes.find(n => n.id === cEdge.to);
      if (!ctlNode) return;
      if (!recs.has(ctlNode.id)) recs.set(ctlNode.id, { controlNode: ctlNode, viaIssues: [] });
      recs.get(ctlNode.id).viaIssues.push(sim);
    });
  });
  return Array.from(recs.values());
}

function submitRecommendation(issueId, selectedControlIds){
  selectedControlIds.forEach(ctlId => {
    ensureCatalogNode(ctlId);
    ensureEdge(issueId, ctlId, { color:{color:'rgba(52,211,163,0.5)', highlight:'#34D3A3'} });
  });
  const gEdge = issueGapEdge(issueId);
  if (gEdge){ edgesDS.remove(issueId + '__' + gEdge.to); nodesDS.remove(gEdge.to); }

  // Now fully "in the network": also connect its own risk taxonomy, same as
  // any already-mapped issue gets by default.
  const rEdge = issueRiskEdge(issueId);
  if (rEdge){ ensureCatalogNode(rEdge.from); ensureEdge(rEdge.from, rEdge.to, { color:{color:'rgba(167,139,250,0.4)', highlight:'#A78BFA'} }); }

  nodesDS.update(Object.assign({ id: issueId }, STYLE.issue));

  resolvedGaps.add(issueId);
  dynamicResolvedCount += 1;
  refreshStats();
}

/* Populate the base network: every already-mapped issue starts already
   connected to its Risk Taxonomy and Control. Gap issues are deliberately
   left with no edges - they start "outside the network" until resolved
   through the theme -> similar issues -> risk/control -> recommend flow. */
GRAPH_DATA.nodes
  .filter(n => n.group === 'issue')
  .forEach(n => revealIssueRiskAndControl(n.id));

function colorDotFor(group){
  return { risk:'var(--risk)', issue:'var(--issue)', issue_gap:'var(--gapAmber)', control:'var(--control)', gap:'var(--gapRed)', theme:'var(--theme)' }[group] || '#888';
}

function field(label, value){
  return `<div class="field"><div class="k">${label}</div><div class="v">${value}</div></div>`;
}

function linkCard(rawNode){
  const sub = { risk:'Risk taxonomy', issue:'Issue', issue_gap:'Issue &middot; gap', control:'Control', gap:'Control gap', theme:'Theme' }[rawNode.group];
  return `<div class="link-card" data-nav="${rawNode.id}">
            <span class="dot" style="background:${colorDotFor(rawNode.group)}"></span>
            <span class="txt">${rawNode.label}<span class="sub mono">${rawNode.id} &middot; ${sub}</span></span>
          </div>`;
}

function activateNode(id){
  ensureCatalogNode(id);
  const node = nodesDS.get(id);
  if (!node) return;
  network.selectNodes([id]);
  network.focus(id, { scale:1.0, animation:{ duration:400 } });
  dispatchClick(node._raw);
}

function attachNavHandlers(){
  body.querySelectorAll('[data-nav]').forEach(el => {
    el.addEventListener('click', () => activateNode(el.getAttribute('data-nav')));
  });
}

function renderIssuePanel(raw){
  const d = raw.details;
  const isGap = raw.group === 'issue_gap' && !resolvedGaps.has(raw.id);
  let html = `<span class="pill issue">Issue</span>`;
  html += `<h3>${d.title}</h3><div class="id mono">${d.id}</div>`;
  html += field('Description', d.description);

  const selected = selectedThemesByIssue[raw.id] || new Set();
  const possible = (d.possible_themes && d.possible_themes.length) ? d.possible_themes : [d.theme];
  html += `<div class="field"><div class="k">${possible.length > 1 ? 'Possible themes - click to tag' : 'Theme - click to tag'}</div><div class="v">`;
  possible.forEach(t => {
    const isSel = selected.has(t);
    html += `<span class="theme-chip ${isSel ? 'selected' : ''}" data-theme="${t}" data-issue="${raw.id}">${isSel ? '<span class="checkmark">&#10003;</span>' : ''}${t}</span>`;
  });
  html += `</div></div>`;

  if (resolvedGaps.has(raw.id)){
    html += `<div class="success-banner">&#10003; Control recommendation submitted for this issue.</div>`;
  } else if (isGap){
    html += `<div class="gap-callout"><div class="k">&#9888; No control mapped</div><div id="recWrap"></div></div>`;
  } else {
    const cEdge = issueControlEdge(raw.id);
    if (cEdge){
      html += `<div class="field"><div class="k">Control status</div><div class="v" style="color:var(--control); font-weight:600;">&#10003; Mapped to a control</div></div>`;
    }
  }

  body.innerHTML = html;
  panel.classList.add('open');

  body.querySelectorAll('.theme-chip').forEach(chip => {
    chip.addEventListener('click', () => {
      toggleThemeForIssue(raw.id, chip.getAttribute('data-theme'), chip);
      if (isGap) renderRecommendations(raw.id);
    });
  });

  if (isGap) renderRecommendations(raw.id);
  attachNavHandlers();
}

function renderRecommendations(issueId){
  const wrap = document.getElementById('recWrap');
  if (!wrap) return;
  const themeSet = selectedThemesByIssue[issueId] || new Set();

  if (themeSet.size === 0){
    wrap.innerHTML = `<div class="recommend-empty">Tag at least one theme above to see recommended controls drawn from similar issues.</div>`;
    return;
  }

  const recs = computeRecommendedControls(issueId, themeSet);
  if (recs.length === 0){
    wrap.innerHTML = `<div class="recommend-empty">No similar issue under the selected theme(s) has a mapped control yet.</div>`;
    return;
  }

  let html = `<div class="recommend-box"><div class="k">Recommended controls</div>`;
  recs.forEach(r => {
    const viaLabel = r.viaIssues.map(i => i.id).join(', ');
    html += `<label class="rec-item">
               <input type="checkbox" class="rec-cb" value="${r.controlNode.id}">
               <span class="txt">${r.controlNode.label}<span class="sub mono">${r.controlNode.id} &middot; via ${viaLabel}</span></span>
             </label>`;
  });
  html += `<button class="btn primary" id="submitRecBtn">Submit selected control(s)</button></div>`;
  wrap.innerHTML = html;

  document.getElementById('submitRecBtn').addEventListener('click', () => {
    const chosen = Array.from(wrap.querySelectorAll('.rec-cb:checked')).map(cb => cb.value);
    if (chosen.length === 0){ alert('Select at least one recommended control before submitting.'); return; }
    submitRecommendation(issueId, chosen);
    renderIssuePanel(nodesDS.get(issueId)._raw);
  });
}

  const themeName = raw.details.name;

  html += `<div class="field"><div class="k">Similar issues (${similar.length})</div></div>`;

  panel.classList.add('open');
  attachNavHandlers();

function renderRiskPanel(raw){

  html += `<h3>${d.name}</h3><div class="id mono">${d.id}</div>`;

  html += `<div class="field"><div class="k">Issues under this risk (${related.length})</div></div>`;
  related.forEach(iss => { html += linkCard(iss); });
  body.innerHTML = html;
  panel.classList.add('open');
  attachNavHandlers();
}

function renderControlPanel(raw){
  const d = raw.details;
  let html = `<span class="pill control">Control</span>`;
  html += `<h3>${d.title}</h3><div class="id mono">${d.id}</div>`;
  html += field('Description', d.description);
  html += `<div class="field"><div class="k">Theme</div><div class="v theme-tag">${d.theme}</div></div>`;

  panel.classList.add('open');

}

function dispatchClick(raw){
  switch(raw.group){
    case 'issue': case 'issue_gap': renderIssuePanel(raw); break;
    case 'theme': renderThemePanel(raw); break;
    case 'risk':  renderRiskPanel(raw); break;
    case 'control': renderControlPanel(raw); break;
    case 'gap': {
      const issueId = raw.details.issue_id;
      activateNode(issueId);
      break;
    }
  }
}

function applyFilters(){
  const activeRisks  = [...document.querySelectorAll('.risk-cb:checked')].map(el => el.value);
  const activeThemes = [...document.querySelectorAll('.theme-cb:checked')].map(el => el.value);
  const gapsOnly = document.getElementById('gapsOnlyToggle').checked;
  const query = document.getElementById('searchInput').value.trim().toLowerCase();

  nodesDS.forEach(n => {
    const raw = n._raw;
    const d = raw.details || {};
    let visible = true;
    const rt = d.risk_taxonomy || d.name || d.linked_risk_taxonomy;
    if (raw.group !== 'risk' && raw.group !== 'theme'){
      if (rt && !activeRisks.includes(rt)) visible = false;
      if (d.theme && !activeThemes.includes(d.theme)) visible = false;
    }
    if (gapsOnly && !(raw.group === 'gap' || raw.group === 'issue_gap')) visible = false;
    if (query){
      const hay = (n.label + ' ' + n.id + ' ' + (d.description||'')).toLowerCase();
      if (!hay.includes(query)) visible = false;
    }
    if (raw.group === 'risk' || raw.group === 'theme') visible = true;
    nodesDS.update({ id: n.id, hidden: !visible });
  });

  edgesDS.forEach(e => {
    const fromNode = nodesDS.get(e.from), toNode = nodesDS.get(e.to);
    const bothVisible = fromNode && toNode && !fromNode.hidden && !toNode.hidden;
    edgesDS.update({ id: e.id, hidden: !bothVisible });
  });
}

document.querySelectorAll('.risk-cb, .theme-cb').forEach(el => el.addEventListener('change', applyFilters));
document.getElementById('gapsOnlyToggle').addEventListener('change', applyFilters);
document.getElementById('searchInput').addEventListener('input', applyFilters);

    .forEach(n => revealIssueRiskAndControl(n.id));

/* =========================================================================
   8. RESILIENT LIBRARY LOADER
   Tries a couple of different CDNs in turn (some networks/firewalls block
   one but not another). Only calls initApp() once vis-network has actually
   loaded; shows a clear message in place of a blank/broken page if every
   source is unreachable, instead of throwing "vis is not defined".
   ========================================================================= */
const VIS_SOURCES = [
  'https://cdnjs.cloudflare.com/ajax/libs/vis-network/10.0.2/standalone/umd/vis-network.min.js',
  'https://cdn.jsdelivr.net/npm/vis-network@10.0.2/standalone/umd/vis-network.min.js',
  'https://unpkg.com/vis-network@10.0.2/standalone/umd/vis-network.min.js',
];

function loadVis(sourceIndex){
  if (sourceIndex >= VIS_SOURCES.length){
    document.getElementById('graph').innerHTML = `
      <div class="cdn-error">
        <div class="box">
          <h3>Graph library couldn't load</h3>
          <p>This page loads its charting library (<code>vis-network</code>) from a public CDN at
          open time. All CDN sources were unreachable from this browser/network just now.</p>
          <p>Things to try: open this file with an active internet connection, check whether your
          network or firewall blocks <code>cdnjs.cloudflare.com</code>, <code>jsdelivr.net</code>, and
          <code>unpkg.com</code>, or try a different browser/network.</p>
        </div>
      </div>`;
    return;
  }
  const s = document.createElement('script');
  s.src = VIS_SOURCES[sourceIndex];
  s.onload = () => {
    if (typeof vis === 'undefined'){ loadVis(sourceIndex + 1); return; }
    try { initApp(); } catch(err){
      console.error('Graph init failed:', err);
      document.getElementById('graph').innerHTML =
        '<div class="cdn-error"><div class="box"><h3>Something went wrong rendering the graph</h3><p>' +
        (err && err.message ? err.message : 'Unknown error') + '</p></div></div>';
    }
  };
  s.onerror = () => loadVis(sourceIndex + 1);
  document.head.appendChild(s);
}

loadVis(0);
