def remove_isolated_controls(llm_controls, similarity_dict, threshold=0.65):
    """
    Remove controls that have no strong connections to other selected controls.
    
    Logic: Real controls typically connect to other relevant controls in the 
    similarity matrix. Isolated controls are likely false positives.
    """
    
    filtered = set()
    
    for ctrl in llm_controls:
        if ctrl not in similarity_dict:
            # Control not in similarity matrix - likely invalid
            continue
        
        # Find how many OTHER selected controls are similar to this one
        neighbor_count = 0
        for other_ctrl in llm_controls:
            if ctrl != other_ctrl:
                if other_ctrl in similarity_dict[ctrl]:
                    sim_score = similarity_dict[ctrl][other_ctrl]
                    if sim_score >= threshold:  # Has strong connection
                        neighbor_count += 1
        
        # Keep if has at least 1 strong neighbor
        if neighbor_count >= 1:
            filtered.add(ctrl)
    
    # Ensure at least 1 control
    if len(filtered) == 0 and len(llm_controls) > 0:
        # Keep the control with most neighbors
        best_ctrl = None
        best_neighbor_count = -1
        
        for ctrl in llm_controls:
            if ctrl not in similarity_dict:
                continue
            neighbor_count = 0
            for other_ctrl in llm_controls:
                if ctrl != other_ctrl and other_ctrl in similarity_dict[ctrl]:
                    if similarity_dict[ctrl][other_ctrl] >= threshold:
                        neighbor_count += 1
            
            if neighbor_count > best_neighbor_count:
                best_neighbor_count = neighbor_count
                best_ctrl = ctrl
        
        if best_ctrl:
            filtered.add(best_ctrl)
    
    return filtered

def remove_weak_connections(llm_controls, similarity_dict, threshold=0.50):
    """
    Remove controls that have low average similarity to other selected controls.
    
    Logic: Controls that fit well with others have high average similarity.
    Controls with low average similarity are likely errors.
    """
    
    if len(llm_controls) <= 1:
        return llm_controls
    
    filtered = set()
    
    for ctrl in llm_controls:
        if ctrl not in similarity_dict:
            continue
        
        # Calculate average similarity to all other controls
        similarities = []
        for other_ctrl in llm_controls:
            if ctrl != other_ctrl and other_ctrl in similarity_dict[ctrl]:
                similarities.append(similarity_dict[ctrl][other_ctrl])
        
        if similarities:
            avg_sim = np.mean(similarities)
            if avg_sim >= threshold:
                filtered.add(ctrl)
    
    # Ensure at least 1 control
    if len(filtered) == 0 and len(llm_controls) > 0:
        # Keep control with highest average similarity
        best_ctrl = None
        best_avg = -1
        
        for ctrl in llm_controls:
            if ctrl not in similarity_dict:
                continue
            similarities = [similarity_dict[ctrl].get(other, 0) for other in llm_controls if other != ctrl]
            if similarities:
                avg_sim = np.mean(similarities)
                if avg_sim > best_avg:
                    best_avg = avg_sim
                    best_ctrl = ctrl
        
        if best_ctrl:
            filtered.add(best_ctrl)
    
    return filtered

def identify_control_clusters(controls, similarity_dict, cluster_threshold=0.70):
    """
    Identify which controls form coherent groups (clusters).
    Remove controls that don't fit into any cluster.
    
    Logic: Real controls form semantic clusters. Isolated controls that 
    don't cluster with others are likely false positives.
    """
    
    if len(controls) <= 2:
        return controls
    
    # For each control, find how many others it's similar to (forms cluster)
    cluster_membership = {}
    
    for ctrl in controls:
        if ctrl not in similarity_dict:
            cluster_membership[ctrl] = 0
            continue
        
        # Count how many controls this one connects to (forms cluster with)
        cluster_size = 0
        for other_ctrl in controls:
            if ctrl != other_ctrl and other_ctrl in similarity_dict[ctrl]:
                sim_score = similarity_dict[ctrl][other_ctrl]
                if sim_score >= cluster_threshold:
                    cluster_size += 1
        
        cluster_membership[ctrl] = cluster_size
    
    # Keep only controls that are in clusters (not isolated)
    filtered = set()
    for ctrl, cluster_size in cluster_membership.items():
        if cluster_size >= 1:  # At least 1 connection
            filtered.add(ctrl)
    
    # Ensure at least 1 control
    if len(filtered) == 0 and len(controls) > 0:
        # Keep the control with largest cluster
        max_ctrl = max(cluster_membership.items(), key=lambda x: x[1])[0]
        filtered.add(max_ctrl)
    
    return filtered

def combine_all_strategies(llm_controls, similarity_dict, 
                          isolation_threshold=0.65,
                          weak_threshold=0.45,
                          cluster_threshold=0.65):
    """
    Apply all 3 strategies sequentially to progressively refine the mapping.
    """
    
    # Apply Strategy 1
    after_s1 = remove_isolated_controls(llm_controls, similarity_dict, isolation_threshold)
    
    # Apply Strategy 2
    after_s2 = remove_weak_connections(after_s1, similarity_dict, weak_threshold)
    
    # Apply Strategy 3
    final = identify_control_clusters(after_s2, similarity_dict, cluster_threshold)
    
    return final
