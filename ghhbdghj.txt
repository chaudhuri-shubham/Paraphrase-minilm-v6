def validate_control_relevance(regulation_text, controls_to_validate, 
                               control_descriptions, batch_size=5):
    """
    Use LLM to validate if each control is actually relevant to the regulation.
    
    Logic: LLM can determine if control is truly relevant or a false positive.
    """
    
    if not controls_to_validate:
        return set()
    
    relevant_controls = set()
    
    # Split controls into batches for efficiency
    control_list = list(controls_to_validate)
    
    # Create validation prompt
    control_info = ""
    for ctrl in control_list:
        desc = control_descriptions.get(ctrl, "No description")
        control_info += f"{ctrl}: {desc}\n"
    
    prompt = f"""Given this regulatory requirement:
"{regulation_text}"

Below are controls that were mapped to it. For EACH control, decide if it's 
truly relevant (YES) or likely a false positive (NO).

Return ONLY comma-separated answers: YES or NO for each control.
Answer format: CONTROL_ID:YES/NO,CONTROL_ID:YES/NO

Controls:
{control_info}

Answer:"""
    
    try:
        response = client.messages.create(
            model="claude-opus-4-5-20251101",
            max_tokens=500,
            messages=[{
                "role": "user",
                "content": prompt
            }]
        )
        
        response_text = response.content[0].text
        
        # Parse response
        for line in response_text.split('\n'):
            if ':' in line:
                parts = line.split(':')
                if len(parts) == 2:
                    ctrl_id = parts[0].strip()
                    answer = parts[1].strip().upper()
                    if answer == 'YES' and ctrl_id in controls_to_validate:
                        relevant_controls.add(ctrl_id)
        
        # Fallback: if parsing failed, return original
        if not relevant_controls:
            relevant_controls = controls_to_validate
    
    except Exception as e:
        print(f"  ⚠ LLM validation error: {str(e)}")
        relevant_controls = controls_to_validate
    
    return relevant_controls

print("\nApplying LLM relevance validation...")

def validate_semantic_relationships(controls, similarity_dict, 
                                   control_descriptions, threshold=0.60):
    """
    For each control, find if there are semantically similar controls in the set.
    Controls that are semantically similar to others are likely correct.
    Isolated controls are likely false positives.
    
    More strict than previous strategies.
    """
    
    if len(controls) <= 1:
        return controls
    
    validated = set()
    
    for ctrl in controls:
        if ctrl not in similarity_dict:
            continue
        
        # Find strongest similarity to ANY other control in set
        max_similarity = 0.0
        for other_ctrl in controls:
            if ctrl != other_ctrl and other_ctrl in similarity_dict[ctrl]:
                sim = similarity_dict[ctrl][other_ctrl]
                max_similarity = max(max_similarity, sim)
        
        # Keep only if has strong relationship
        if max_similarity >= threshold:
            validated.add(ctrl)
    
    # Ensure at least 1 control
    if len(validated) == 0 and len(controls) > 0:
        # Keep control with strongest similarity to any other
        best_ctrl = None
        best_sim = -1
        
        for ctrl in controls:
            if ctrl not in similarity_dict:
                continue
            max_sim = 0.0
            for other_ctrl in controls:
                if ctrl != other_ctrl and other_ctrl in similarity_dict[ctrl]:
                    max_sim = max(max_sim, similarity_dict[ctrl][other_ctrl])
            
            if max_sim > best_sim:
                best_sim = max_sim
                best_ctrl = ctrl
        
        if best_ctrl:
            validated.add(best_ctrl)
    
    return validated

print("\nApplying semantic relationship validation...")

def check_domain_coherence(controls, similarity_dict, control_descriptions):
    """
    Check if all controls form a coherent domain/topic.
    
    Logic: Calculate average pairwise similarity between all controls.
    If average is high, they form coherent group (likely correct).
    If average is low, they're from different domains (likely has FPs).
    """
    
    if len(controls) <= 1:
        return controls
    
    # Calculate all pairwise similarities
    pairwise_similarities = []
    controls_list = list(controls)
    
    for i, ctrl1 in enumerate(controls_list):
        if ctrl1 not in similarity_dict:
            continue
        for ctrl2 in controls_list[i+1:]:
            if ctrl2 in similarity_dict[ctrl1]:
                sim = similarity_dict[ctrl1][ctrl2]
                pairwise_similarities.append(sim)
    
    if not pairwise_similarities:
        return controls
    
    avg_pairwise_sim = np.mean(pairwise_similarities)
    
    # If average pairwise similarity is low, has false positives
    # Remove controls that deviate from the group
    
    coherence_threshold = 0.55
    
    if avg_pairwise_sim < coherence_threshold:
        # Group is incoherent, remove outliers
        validated = set()
        
        for ctrl in controls:
            if ctrl not in similarity_dict:
                continue
            
            # Calculate how similar this control is to others
            ctrl_similarities = []
            for other_ctrl in controls:
                if ctrl != other_ctrl and other_ctrl in similarity_dict[ctrl]:
                    ctrl_similarities.append(similarity_dict[ctrl][other_ctrl])
            
            if ctrl_similarities:
                ctrl_avg_sim = np.mean(ctrl_similarities)
                # Keep if closer to average than 0.5
                if ctrl_avg_sim >= 0.50:
                    validated.add(ctrl)
        
        # Ensure at least 1
        if len(validated) == 0:
            validated = controls
        
        return validated
    else:
        # Group is coherent, keep all
        return controls

print("\nApplying domain coherence check...")

def analyze_control_density(controls, similarity_dict):
    """
    Analyze the "density" of the control set.
    
    Logic: Controls form dense clusters in similarity space.
    Very sparse sets likely have false positives.
    """
    
    if len(controls) <= 2:
        return controls
    
    # Calculate density = average number of neighbors for each control
    density_scores = []
    
    for ctrl in controls:
        if ctrl not in similarity_dict:
            density_scores.append(0)
            continue
        
        # Count neighbors (other controls with high similarity)
        neighbors = 0
        for other_ctrl in controls:
            if ctrl != other_ctrl and other_ctrl in similarity_dict[ctrl]:
                if similarity_dict[ctrl][other_ctrl] >= 0.65:
                    neighbors += 1
        
        density_scores.append(neighbors)
    
    # Remove controls with 0 neighbors (isolated)
    validated = set()
    for ctrl, density in zip(controls, density_scores):
        if density > 0:
            validated.add(ctrl)
    
    # Ensure at least 1
    if len(validated) == 0:
        validated = controls
    
    return validated

def validate_with_keywords(regulation_text, controls, control_descriptions):
    """
    Check if control descriptions contain keywords from regulation text.
    
    Logic: Correct controls should share vocabulary with regulation.
    False positives often have no keyword overlap.
    """
    
    # Extract key words from regulation (>3 chars, not common words)
    common_words = {'the', 'and', 'for', 'must', 'will', 'are', 'be', 'to', 
                   'in', 'is', 'of', 'all', 'a', 'or', 'an', 'by', 'with'}
    
    reg_words = set()
    for word in regulation_text.lower().split():
        # Clean word
        clean_word = ''.join(c for c in word if c.isalnum())
        if len(clean_word) > 3 and clean_word not in common_words:
            reg_words.add(clean_word)
    
    validated = set()
    
    for ctrl in controls:
        if ctrl not in control_descriptions:
            validated.add(ctrl)
            continue
        
        # Check keyword overlap
        ctrl_desc = control_descriptions[ctrl].lower()
        overlap_count = 0
        
        for word in reg_words:
            if word in ctrl_desc:
                overlap_count += 1
        
        # Keep if has keyword overlap (>0)
        if overlap_count > 0:
            validated.add(ctrl)
    
    # Ensure at least 1
    if len(validated) == 0:
        validated = controls
    
    return validated
