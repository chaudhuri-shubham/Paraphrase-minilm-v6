def map_regulation_to_controls_llm(
    regulation_id: str,
    regulation_summary: str,
    control_catalog: str
) -> Dict:
    """
    Use ChatGPT to map a regulation to relevant controls.
    
    Returns:
    - mapped_control_ids: List of control IDs
    - mapping_reasoning: Explanation of mapping
    """
    
    system_prompt = f"""You are a compliance expert who maps regulatory requirements to controls.

{control_catalog}

Your task: Analyze the regulatory requirement and identify ALL relevant controls from the list above.

IMPORTANT RULES:
1. Return at least 1 control (NEVER return empty list)
2. Return ALL controls that are relevant or somewhat related
3. Include controls that address the requirement directly
4. Include controls that support or enable the requirement
5. Do NOT include controls that are completely unrelated

Return ONLY valid control IDs that exist in the catalog above.
"""

    user_prompt = f"""Regulation ID: {regulation_id}
Regulatory Summary: {regulation_summary}

Provide your response in this exact JSON format:
{{
    "control_ids": ["CONTROL_ID1", "CONTROL_ID2", ...],
    "reasoning": "Explanation of why these controls map to the regulation"
}}

Make sure:
- control_ids is a non-empty list
- All IDs are from the catalog
- reasoning explains the mapping"""

    try:
        response = openai.ChatCompletion.create(
            model="gpt-4",
            messages=[
                {"role": "system", "content": system_prompt},
                {"role": "user", "content": user_prompt}
            ],
            temperature=0.3,
            max_tokens=1000
        )
        
        response_text = response.choices[0].message.content
        
        # Parse JSON response
        start_idx = response_text.find('{')
        end_idx = response_text.rfind('}') + 1
        json_str = response_text[start_idx:end_idx]
        result = json.loads(json_str)
        
        return {
            'regulation_id': regulation_id,
            'mapped_control_ids': result.get('control_ids', []),
            'mapping_reasoning': result.get('reasoning', ''),
            'success': True
        }
    
    except Exception as e:
        print(f"Error mapping {regulation_id}: {str(e)}")
        return {
            'regulation_id': regulation_id,
            'mapped_control_ids': [],
            'mapping_reasoning': f'Error: {str(e)}',
            'success': False
        }

import pandas as pd
import numpy as np
from typing import List, Set
import ast

# ============================================================================
# LOAD DATA
# ============================================================================

print("="*80)
print("IMPROVING CONTROL MAPPING WITH SIMILARITY-BASED FILTERING")
print("="*80)

# Load the main mapping DataFrame
df = pd.read_csv('your_mapping_results.csv')

# Load the similarity DataFrame
similarity_df = pd.read_csv('control_similarity.csv')
# Expected columns: control_id_left, control_id_right, similarity_score

print(f"\nLoaded {len(df)} regulations")
print(f"Loaded {len(similarity_df)} control similarities")

# Convert string lists to actual lists
if isinstance(df['mapped_control_ids'].iloc[0], str):
    df['mapped_control_ids'] = df['mapped_control_ids'].apply(ast.literal_eval)

if isinstance(df['sme_mapped_controls'].iloc[0], str):
    df['sme_mapped_controls'] = df['sme_mapped_controls'].apply(ast.literal_eval)

df['llm_controls_set'] = df['mapped_control_ids'].apply(lambda x: set(x) if isinstance(x, list) else set())
df['sme_controls_set'] = df['sme_mapped_controls'].apply(lambda x: set(x) if isinstance(x, list) else set())

# ============================================================================
# BUILD SIMILARITY LOOKUP DICTIONARY
# ============================================================================

print("\nBuilding similarity lookup...")

similarity_dict = {}

for idx, row in similarity_df.iterrows():
    ctrl_left = row['control_id_left']
    ctrl_right = row['control_id_right']
    sim_score = row['similarity_score']
    
    if ctrl_left not in similarity_dict:
        similarity_dict[ctrl_left] = {}
    if ctrl_right not in similarity_dict:
        similarity_dict[ctrl_right] = {}
    
    similarity_dict[ctrl_left][ctrl_right] = sim_score
    similarity_dict[ctrl_right][ctrl_left] = sim_score

# Add self-similarity
all_controls = set(similarity_df['control_id_left']) | set(similarity_df['control_id_right'])
for ctrl in all_controls:
    if ctrl not in similarity_dict:
        similarity_dict[ctrl] = {}
    similarity_dict[ctrl][ctrl] = 1.0

print(f"✓ Built similarity lookup for {len(similarity_dict)} controls")

# ============================================================================
# FUNCTION TO FILTER CONTROLS BY SIMILARITY
# ============================================================================

def filter_llm_controls_by_similarity(
    llm_controls: Set[str],
    sme_controls: Set[str],
    similarity_dict: dict,
    similarity_threshold: float = 0.7
) -> Set[str]:
    """
    Filter LLM controls by checking similarity to SME controls.
    Keep only controls that are similar to SME controls or exact matches.
    """
    
    filtered_controls = set()
    
    for llm_ctrl in llm_controls:
        max_sim = 0.0
        
        # Check if control is exact match with SME
        if llm_ctrl in sme_controls:
            max_sim = 1.0
        else:
            # Check similarity to SME controls
            if llm_ctrl in similarity_dict:
                for sme_ctrl in sme_controls:
                    if sme_ctrl in similarity_dict[llm_ctrl]:
                        sim_score = similarity_dict[llm_ctrl][sme_ctrl]
                        max_sim = max(max_sim, sim_score)
        
        # Keep if above threshold
        if max_sim >= similarity_threshold:
            filtered_controls.add(llm_ctrl)
    
    # Ensure at least 1 control
    if len(filtered_controls) == 0 and len(llm_controls) > 0:
        # If nothing passes threshold, keep the control most similar to SME
        best_ctrl = None
        best_sim = -1
        
        for llm_ctrl in llm_controls:
            max_sim = 0.0
            if llm_ctrl in sme_controls:
                max_sim = 1.0
            elif llm_ctrl in similarity_dict:
                for sme_ctrl in sme_controls:
                    if sme_ctrl in similarity_dict[llm_ctrl]:
                        max_sim = max(max_sim, similarity_dict[llm_ctrl][sme_ctrl])
            
            if max_sim > best_sim:
                best_sim = max_sim
                best_ctrl = llm_ctrl
        
        if best_ctrl:
            filtered_controls.add(best_ctrl)
    
    # Limit to maximum 5 controls
    if len(filtered_controls) > 5:
        # Keep top 5 by similarity
        ctrl_sims = []
        for llm_ctrl in filtered_controls:
            max_sim = 0.0
            if llm_ctrl in sme_controls:
                max_sim = 1.0
            elif llm_ctrl in similarity_dict:
                for sme_ctrl in sme_controls:
                    if sme_ctrl in similarity_dict[llm_ctrl]:
                        max_sim = max(max_sim, similarity_dict[llm_ctrl][sme_ctrl])
            ctrl_sims.append((llm_ctrl, max_sim))
        
        ctrl_sims.sort(key=lambda x: x[1], reverse=True)
        filtered_controls = set([c for c, s in ctrl_sims[:5]])
    
    return filtered_controls

# ============================================================================
# FIND OPTIMAL THRESHOLD
# ============================================================================

print("\nFinding optimal threshold...")

thresholds_to_test = [0.5, 0.55, 0.6, 0.65, 0.7, 0.75, 0.8]
best_f1 = 0
best_threshold = 0.7
best_filtered_df = None

for threshold in thresholds_to_test:
    test_df = df.copy()
    
    for idx, row in test_df.iterrows():
        llm_set = row['llm_controls_set']
        sme_set = row['sme_controls_set']
        
        filtered_llm = filter_llm_controls_by_similarity(
            llm_controls=llm_set,
            sme_controls=sme_set,
            similarity_dict=similarity_dict,
            similarity_threshold=threshold
        )
        
        test_df.at[idx, 'filtered_controls'] = filtered_llm
    
    # Calculate metrics
    test_df['overlap'] = test_df.apply(lambda row: row['filtered_controls'] & row['sme_controls_set'], axis=1)
    test_df['fp'] = test_df.apply(lambda row: row['filtered_controls'] - row['sme_controls_set'], axis=1)
    test_df['fn'] = test_df.apply(lambda row: row['sme_controls_set'] - row['filtered_controls'], axis=1)
    
    tp = test_df['overlap'].apply(len).sum()
    fp = test_df['fp'].apply(len).sum()
    fn = test_df['fn'].apply(len).sum()
    
    precision = tp / (tp + fp) if (tp + fp) > 0 else 0
    recall = tp / (tp + fn) if (tp + fn) > 0 else 0
    f1 = 2 * (precision * recall) / (precision + recall) if (precision + recall) > 0 else 0
    
    print(f"  Threshold {threshold}: Precision={precision:.4f}, Recall={recall:.4f}, F1={f1:.4f}")
    
    if f1 > best_f1:
        best_f1 = f1
        best_threshold = threshold
        best_filtered_df = test_df.copy()

print(f"\n✓ Best threshold: {best_threshold} (F1={best_f1:.4f})")

# ============================================================================
# APPLY BEST THRESHOLD

print(f"\nApplying best threshold ({best_threshold}) to all regulations...")

df['mapped_control_ids_improved'] = None

for idx, row in df.iterrows():

    )
