# Format controls as a readable catalog for the LLM
control_catalog = "AVAILABLE CONTROLS:\n" + "="*80 + "\n\n"

print("="*80)

def map_regulation_to_controls_llm(
    regulation_id: str,
    regulation_summary: str,
    control_catalog: str
) -> Dict:
    """
    Use ChatGPT to map a regulation to relevant controls.
    
    Returns:
    - mapped_control_ids: List of control IDs
    - mapping_reasoning: Explanation of mapping
    """
    
    system_prompt = f"""You are a compliance expert who maps regulatory requirements to controls.

{control_catalog}

Your task: Analyze the regulatory requirement and identify ALL relevant controls from the list above.

IMPORTANT RULES:
1. Return at least 1 control (NEVER return empty list)
2. Return ALL controls that are relevant or somewhat related
3. Include controls that address the requirement directly
4. Include controls that support or enable the requirement
5. Do NOT include controls that are completely unrelated

Return ONLY valid control IDs that exist in the catalog above.
"""

    user_prompt = f"""Regulation ID: {regulation_id}
Regulatory Summary: {regulation_summary}

Provide your response in this exact JSON format:
{{
    "control_ids": ["CONTROL_ID1", "CONTROL_ID2", ...],
    "reasoning": "Explanation of why these controls map to the regulation"
}}

Make sure:
- control_ids is a non-empty list
- All IDs are from the catalog
- reasoning explains the mapping"""

    try:
        response = openai.ChatCompletion.create(
            model="gpt-4",
            messages=[
                {"role": "system", "content": system_prompt},
                {"role": "user", "content": user_prompt}
            ],
            temperature=0.3,
            max_tokens=1000
        )
        
        response_text = response.choices[0].message.content
        
        # Parse JSON response
        start_idx = response_text.find('{')
        end_idx = response_text.rfind('}') + 1
        json_str = response_text[start_idx:end_idx]
        result = json.loads(json_str)
        
        return {
            'regulation_id': regulation_id,
            'mapped_control_ids': result.get('control_ids', []),
            'mapping_reasoning': result.get('reasoning', ''),
            'success': True
        }
    
    except Exception as e:
        print(f"Error mapping {regulation_id}: {str(e)}")
        return {
            'regulation_id': regulation_id,
            'mapped_control_ids': [],
            'mapping_reasoning': f'Error: {str(e)}',
            'success': False
        }

# STEP 4: MAP ALL REGULATIONS
# ============================================================================

print("\nMapping regulations to controls...\n")

all_mappings = []

for idx, row in regulatory_df.iterrows():
    regulation_id = row['regulation_id']
    regulation_summary = row['regulatory_summary']
    
    # Get mapping from LLM
    mapping = map_regulation_to_controls_llm(
        regulation_id=regulation_id,
        regulation_summary=regulation_summary,
        control_catalog=control_catalog
    )
    
    all_mappings.append(mapping)
    
    if (idx + 1) % 5 == 0 or (idx + 1) == len(regulatory_df):
        print(f"✓ Processed {idx + 1}/{len(regulatory_df)} regulations")
    
    # Show details
    if mapping['success']:
        print(f"  {regulation_id} → {mapping['mapped_control_ids']} ({len(mapping['mapped_control_ids'])} controls)")

# Convert mappings to DataFrame

# Add additional columns

# Merge with original regulatory data
results_df = results_df.merge(
    regulatory_df,
    on='regulation_id',
    how='left'
)

print("\nResults DataFrame:")
print(results_df[['regulation_id', 'regulatory_summary', 'mapped_control_ids', 'num_controls_mapped']].head(10))

print(f"  Average controls per regulation: {results_df['num_controls_mapped'].mean():.2f}")
print(f"  Min controls: {results_df['num_controls_mapped'].min()}")

    regulation_id = row['regulation_id']
    regulation_summary = row['regulatory_summary']
    mapped_controls = row['mapped_control_ids']

        # Find control description

        if len(control_desc) > 0:
            expanded_rows.append({

                'regulatory_summary': regulation_summary,
                'control_id': control_id,
                'control_description': control_desc[0],
                'mapping_reasoning': reasoning

expanded_df = pd.DataFrame(expanded_rows)
