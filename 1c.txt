import pandas as pd
import numpy as np
from openai import OpenAI
from collections import defaultdict

# Initialize OpenAI client
client = OpenAI(api_key="your-openai-api-key")

print("="*80)
print("REGULATORY TEXT TO CONTROL MAPPING - SIMPLIFIED")
print("Using GPT-4o Mini with Direct LLM-based Mapping")
print("="*80)

# ============================================================================
# LOAD DATA
# ============================================================================

print("\nLoading data...")

regulatory_df = pd.read_csv('regulatory_texts.csv')
controls_df = pd.read_csv('controls_catalog.csv')

print(f"  Regulatory Texts: {len(regulatory_df)}")
print(f"  Control Catalog: {len(controls_df)} controls")

# Build control catalog
control_catalog = {}
for _, row in controls_df.iterrows():
    ctrl_id = row['control_id']
    title = row['control_title']
    desc = row['control_description']
    control_catalog[ctrl_id] = f"{ctrl_id}: {title}"

print(f"✓ Control catalog built: {len(control_catalog)} controls")

# ============================================================================
# BATCH CONTROLS FOR LLM (Send in batches of 100)
# ============================================================================

print("\n" + "="*80)
print("Preparing control batches for LLM processing...")
print("="*80)

control_ids = list(control_catalog.keys())
batch_size = 100

control_batches = []
for i in range(0, len(control_ids), batch_size):
    batch = control_ids[i:i+batch_size]
    control_batches.append(batch)

print(f"✓ Created {len(control_batches)} batches of {batch_size} controls each")

# ============================================================================
# MAIN MAPPING FUNCTION
# ============================================================================

def map_regulation_to_controls(regulation_id, regulation_title, regulation_text, 
                               control_batches, control_catalog):
    """
    Use LLM to map regulation to controls.
    Process in batches and get top matches.
    """
    
    all_matched_controls = []
    
    for batch_num, batch in enumerate(control_batches):
        # Create control list for this batch
        control_list = ""
        for ctrl_id in batch:
            control_list += f"- {control_catalog[ctrl_id]}\n"
        
        prompt = f"""You are a compliance expert mapping regulatory requirements to security controls.

REGULATORY REQUIREMENT:
Title: {regulation_title}
Text: {regulation_text}

AVAILABLE CONTROLS IN THIS BATCH:
{control_list}

TASK:
1. Identify which controls from this batch are RELEVANT to the regulatory requirement
2. Be INCLUSIVE - include all controls that could help implement/support this regulation
3. Include preventive, detective, corrective controls
4. Include both direct implementations and supporting controls

RETURN FORMAT:
List only the matching control IDs, one per line. Example:
L1C-001
L1C-045
L1C-102

If no controls match, return: NONE"""

        try:
            response = client.messages.create(
                model="gpt-4o-mini",
                max_tokens=500,
                temperature=0.3,
                messages=[
                    {
                        "role": "user",
                        "content": prompt
                    }
                ]
            )
            
            response_text = response.content[0].text.strip()
            
            # Parse response
            for line in response_text.split('\n'):
                line = line.strip().upper()
                if line and line != 'NONE' and line in control_ids:
                    all_matched_controls.append(line)
            
            print(f"  Batch {batch_num + 1}/{len(control_batches)}: Found {len([l for l in response_text.split('\n') if l.strip() and l.strip() != 'NONE' and l.strip().upper() in control_ids])} controls")
        
        except Exception as e:
            print(f"  ⚠ Error processing batch {batch_num + 1}: {str(e)}")
    
    # Remove duplicates and return
    unique_controls = list(set(all_matched_controls))
    return unique_controls

# ============================================================================
# MAP ALL REGULATIONS
# ============================================================================

print("\n" + "="*80)
print("MAPPING REGULATIONS TO CONTROLS")
print("="*80)

all_mappings = []

for idx, row in regulatory_df.iterrows():
    reg_id = row['regulation_id']
    reg_title = row['regulatory_title']
    reg_text = row['regulatory_summary']
    
    print(f"\n[{idx + 1}/{len(regulatory_df)}] Mapping: {reg_id}")
    
    mapped_controls = map_regulation_to_controls(
        reg_id, 
        reg_title, 
        reg_text,
        control_batches,
        control_catalog
    )
    
    all_mappings.append({
        'regulation_id': reg_id,
        'regulation_title': reg_title,
        'regulation_text': reg_text[:150],
        'mapped_control_ids': ','.join(sorted(mapped_controls)),
        'control_count': len(mapped_controls)
    })
    
    print(f"  ✓ Mapped {len(mapped_controls)} controls")

# ============================================================================
# RESULTS
# ============================================================================

print("\n" + "="*80)
print("MAPPING RESULTS")
print("="*80)

output_df = pd.DataFrame(all_mappings)

print(f"\nSummary:")
print(f"  Total Regulations: {len(output_df)}")
print(f"  Avg Controls per Regulation: {output_df['control_count'].mean():.1f}")
print(f"  Min/Max: {output_df['control_count'].min()} / {output_df['control_count'].max()}")

print(f"\nControl Count Distribution:")
print(output_df['control_count'].describe())

# Save results
output_df.to_csv('regulatory_control_mapping_final.csv', index=False)
print("\n✓ Saved: regulatory_control_mapping_final.csv")

print("\n" + "="*80)
print("MAPPING COMPLETE")
