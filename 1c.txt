def extract_regulation_keywords(regulation_text, regulation_title, min_length=3):
    """Extract important keywords from regulation."""
    
    keywords = set()
    
    # Title keywords (higher priority)
    title_words = str(regulation_title).lower().split()
    keywords.update(title_words)
    
    # Summary keywords
    summary_words = str(regulation_text).lower().split()
    keywords.update(summary_words)
    
    # Remove common words and short words
    common_words = {'the', 'a', 'an', 'and', 'or', 'is', 'are', 'be', 'to', 'of', 'in', 'for', 'with', 'by', 'on', 'at', 'this', 'that', 'which', 'who', 'should', 'must', 'shall', 'will', 'can', 'may', 'could', 'would', 'not', 'no', 'yes', 'or', 'it', 'its', 'as', 'from', 'do', 'does', 'did', 'have', 'has', 'had', 'all', 'some', 'any', 'each', 'every', 'both', 'either', 'neither', 'such', 'what', 'when', 'where', 'why', 'how', 'if', 'unless', 'until', 'while', 'during', 'before', 'after', 'above', 'below', 'between', 'through', 'throughout', 'within', 'without', 'about', 'around', 'over', 'under', 'out', 'up', 'down', 'off', 'more', 'less', 'most', 'least', 'many', 'few', 'much', 'little', 'so', 'than', 'are', 'were', 'been', 'being', 'my', 'your', 'his', 'her', 'our', 'their', 'me', 'you', 'him', 'her', 'us', 'them', 'I', 'we', 'they', 'she', 'he', 'that', 'these', 'those', 'ones', 'other', 'another', 'same', 'different', 'new', 'old', 'good', 'bad', 'better', 'worse', 'best', 'worst', 'high', 'low', 'first', 'last', 'next', 'previous', 'only', 'just', 'also', 'even', 'very', 'too', 'so', 'as', 'because', 'since', 'although', 'though', 'however', 'therefore', 'thus', 'hence', 'otherwise', 'moreover', 'furthermore', 'besides', 'instead', 'rather', 'except', 'but', 'yet', 'nor', 'neither'}
    keywords = {kw for kw in keywords if len(kw) > min_length and kw not in common_words}
    
    return keywords

def direct_keyword_match(regulation_text, regulation_title, control_keywords, top_k=15):
    """Find controls by matching keywords in regulation with control keywords."""
    
    # Extract keywords from regulation
    reg_keywords = extract_regulation_keywords(regulation_text, regulation_title)
    
    if not reg_keywords:
        return set()
    
    # Score each control based on keyword overlap
    control_scores = defaultdict(int)
    
    for ctrl_id, ctrl_kw in control_keywords.items():
        overlap = len(reg_keywords & ctrl_kw)
        if overlap > 0:
            control_scores[ctrl_id] = overlap
    
    # Get top-K controls
    sorted_controls = sorted(control_scores.items(), key=lambda x: x[1], reverse=True)
    top_controls = [ctrl_id for ctrl_id, score in sorted_controls[:top_k]]
    
    return set(top_controls)

def identify_relevant_domains_with_llm(regulation_id, regulation_text, regulation_title, 
                                       domains_list, direct_controls, control_catalog):
    """
    Use LLM to identify which domains of controls are relevant to this regulation.
    """
    
    # Get controls for direct matches to show LLM the style
    direct_control_examples = ""
    for ctrl_id in list(direct_controls)[:3]:
        if ctrl_id in control_catalog:
            ctrl_info = control_catalog[ctrl_id]
            direct_control_examples += f"- {ctrl_id}: {ctrl_info['title']}\n"
    
    # Create domain list with counts
    domain_list_str = ""
    for i, domain in enumerate(domains_list[:50], 1):  # Top 50 domains
        count = len(domains_to_controls[domain])
        domain_list_str += f"{i}. {domain} ({count} controls)\n"
    
    system_prompt = """You are a compliance expert identifying relevant control domains.

Your task: Given a regulatory requirement, identify ALL control domains that are relevant.

Approach:
1. Understand the regulatory requirement deeply
2. Think about what types of controls would be needed
3. Identify ALL potentially relevant domains
4. Be comprehensive - include preventive, detective, and corrective controls
5. Include supporting and enabling controls

Return: List of relevant domain numbers/names"""

    user_prompt = f"""Regulatory Requirement:
Title: {regulation_title}
Text: {regulation_text}

Examples of directly matched controls:
{direct_control_examples}

Available Control Domains:
{domain_list_str}

Which control domains are relevant to this regulation?
List ALL domains that apply (be comprehensive, not restrictive).

Return as: Domain names or numbers, one per line."""

    try:
        response = client.messages.create(
            model="gpt-4o-mini",
            max_tokens=800,
            temperature=0.3,  # Lower temperature for consistency
            messages=[
                {
                    "role": "user",
                    "content": user_prompt
                }
            ],
            system=system_prompt
        )
        
        response_text = response.content[0].text.strip()
        
        # Parse response to extract domain names
        relevant_domains = set()
        
        for line in response_text.split('\n'):
            line = line.strip().lower()
            if len(line) > 2:
                # Remove numbering/bullets
                line = line.lstrip('0123456789.-) ')
                
                # Find matching domains
                for domain in domains_list:
                    if domain.lower() in line or line in domain.lower():
                        relevant_domains.add(domain)
        
        return relevant_domains
    
    except Exception as e:
        print(f"  ⚠ LLM domain identification error: {str(e)}")
        return set()

def expand_from_domains(relevant_domains, domains_to_controls, direct_controls, top_per_domain=4):
    """Get controls from relevant domains."""
    
    expanded = set(direct_controls)
    
    for domain in relevant_domains:
        if domain in domains_to_controls:
            domain_controls = list(domains_to_controls[domain])
            # Add top controls from this domain
            expanded.update(domain_controls[:top_per_domain])
    
    return expanded

def rank_controls_with_llm(regulation_id, regulation_text, regulation_title, 
                           candidate_controls, control_catalog, target_count=6):
    """
    Use LLM to rank candidate controls by relevance and select top ones.
    """
    
    if not candidate_controls:
        return set()
    
    # Prepare control information for LLM
    control_info = "Candidate Controls:\n"
    for i, ctrl_id in enumerate(sorted(candidate_controls)[:20], 1):  # Limit to 20 for token efficiency
        if ctrl_id in control_catalog:
            title = control_catalog[ctrl_id]['title']
            desc = control_catalog[ctrl_id]['description'][:100]  # Truncate description
            control_info += f"{i}. {ctrl_id}: {title}\n"
    
    system_prompt = """You are a compliance expert ranking controls by relevance.

Your task: Rank controls by how relevant they are to the regulation.

Criteria:
- Direct relevance to regulatory requirement
- Importance for compliance
- Practicality and implementation
- Coverage of key regulatory aspects

Return: Top 6 control IDs in order of relevance"""

    user_prompt = f"""Regulatory Requirement:
Title: {regulation_title}
Text: {regulation_text}

{control_info}

Rank the top {target_count} most relevant controls (most relevant first).
Return only control IDs, one per line."""

    try:
        response = client.messages.create(
            model="gpt-4o-mini",
            max_tokens=300,
            temperature=0.2,
            messages=[
                {
                    "role": "user",
                    "content": user_prompt
                }
            ],
            system=system_prompt
        )
        
        response_text = response.content[0].text.strip()
        
        # Parse response to extract control IDs
        ranked_controls = []
        
        for line in response_text.split('\n'):
            line = line.strip()
            # Extract control ID (usually first token)
            ctrl_id = line.split()[0].upper() if line else None
            
            if ctrl_id and ctrl_id in candidate_controls:
                ranked_controls.append(ctrl_id)
        
        # Return top controls
        return set(ranked_controls[:target_count])
    
    except Exception as e:
        print(f"  ⚠ LLM ranking error: {str(e)}")
        # Fallback: return first N controls
        return set(list(candidate_controls)[:target_count])

def calculate_control_confidence(control_id, regulation_text, regulation_title, 
                                control_catalog, direct_matched, domain_matched, llm_ranked):
    """
    Calculate confidence score for a control based on multiple signals.
    """
    
    score = 0.0
    
    # Factor 1: Was it in direct keyword matches? (0.3 weight)
    if control_id in direct_matched:
        score += 0.3
    
    # Factor 2: Was it in domain matches? (0.3 weight)
    if control_id in domain_matched:
        score += 0.3
    
    # Factor 3: Was it in LLM ranking? (0.4 weight)
    if control_id in llm_ranked:
        score += 0.4
    
    # Normalize to 0-1 scale
    confidence = min(score, 1.0)
    
    return confidence

def final_selection(control_set, confidence_scores, max_controls=8, min_confidence=0.0):
    """
    Final selection: Keep controls sorted by confidence.
    """
    
    if not control_set:
        return set()
    
    # Sort by confidence
    sorted_controls = sorted(
        control_set,
        key=lambda x: confidence_scores.get(x, 0.0),
        reverse=True
    )
    
    # Keep top controls with minimum confidence
    selected = []
    for ctrl_id in sorted_controls:
        if len(selected) >= max_controls:
            break
        if confidence_scores.get(ctrl_id, 0.0) >= min_confidence:
            selected.append(ctrl_id)
    
    # If not enough controls, add more regardless of confidence
    if len(selected) < max_controls:
        for ctrl_id in sorted_controls:
            if ctrl_id not in selected:
                selected.append(ctrl_id)
                if len(selected) >= max_controls:
                    break
    
    return set(selected)
