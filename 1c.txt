import pandas as pd
from sentence_transformers import SentenceTransformer, util
from openai import OpenAI

# Initialize clients
client = OpenAI(api_key="YOUR_OPENAI_API_KEY")
embed_model = SentenceTransformer('all-MiniLM-L6-v2')

def get_llm_response(prompt):
    response = client.chat.completions.create(
        model="gpt-4o",
        messages=[{"role": "system", "content": "You are a GRC expert. Your goal is high-recall mapping."},
                  {"role": "user", "content": prompt}],
        temperature=0.1
    )
    return response.choices[0].message.content

# --- STEP 1: Intent Expansion (LLM) ---
def expand_regulation(reg_id, reg_title, reg_desc):
    """Breaks a single regulation into 6 searchable intents for broader coverage."""
    prompt = f"""
    Regulation ID: {reg_id}
    Title: {reg_title}
    Description: {reg_desc}

    Identify 6 distinct security/compliance requirements mentioned or implied above. 
    Each should be a short, technical search query (e.g., 'Encryption at rest' or 'MFA for admins').
    Return only the queries, one per line.
    """
    queries = get_llm_response(prompt).strip().split('\n')
    # Clean up any potential numbering from LLM
    return [q.strip("- 123456. ") for q in queries if q.strip()]

# --- STEP 2 & 3: Retrieval and Mapping Selection ---
def map_regulations_to_controls(reg_df, control_df):
    # Prepare Control Embeddings
    print(f"Embedding {len(control_df)} controls...")
    control_df['full_text'] = control_df['id'] + ": " + control_df['title'] + " " + control_df['description']
    control_embeddings = embed_model.encode(control_df['full_text'].tolist(), convert_to_tensor=True)
    
    mapping_results = []

    for _, reg in reg_df.iterrows():
        reg_id = reg['id']
        print(f"Mapping Regulation ID: {reg_id}...")

        # 1. EXPAND: Get multiple intents
        queries = expand_regulation(reg_id, reg['title'], reg['description'])
        
        # 2. RETRIEVE: Find candidates using vector search
        candidate_indices = set()
        for q in queries:
            q_emb = embed_model.encode(q, convert_to_tensor=True)
            # Fetch Top 15 per query to ensure high coverage
            hits = util.semantic_search(q_emb, control_embeddings, top_k=15)[0]
            for h in hits:
                candidate_indices.add(h['corpus_id'])
        
        candidates = control_df.iloc[list(candidate_indices)]
        
        # 3. SELECT: Final LLM filter for high recall
        # We format the list so the LLM explicitly sees the IDs
        candidate_context = ""
        for _, c in candidates.iterrows():
            candidate_context += f"CONTROL_ID: {c['id']} | TITLE: {c['title']} | DESC: {c['description'][:200]}\n"

        selection_prompt = f"""
        REGULATION_ID: {reg_id}
        REGULATION_TEXT: {reg['description']}

        CANDIDATE CONTROLS:
        {candidate_context}

        TASK:
        From the list above, select ALL Control IDs that address any part of this regulation.
        - Prioritize COVERAGE: if it's a partial match, include it.
        - Return ONLY a comma-separated list of the Control IDs.
        - If no controls match, return 'NONE'.
        """
        
        selected_ids_str = get_llm_response(selection_prompt)
        
        # Store result for this specific Regulation ID
        mapping_results.append({
            "regulation_id": reg_id,
            "mapped_control_ids": selected_ids_str
        })

    return pd.DataFrame(mapping_results)

def map_regulation_with_confidence(regulation_id, regulation_title, regulation_text, 
                                   control_batches, control_catalog, top_k=10):
    """
    Single LLM function that:
    1. Identifies matching controls
    2. Assigns confidence scores
    3. Returns top 10 by confidence
    """
    
    all_controls_with_scores = {}
    
    for batch_num, batch in enumerate(control_batches):
        # Create control list for this batch
        control_list = ""
        for ctrl_id in batch:
            control_list += f"- {control_catalog[ctrl_id]}\n"
        
        prompt = f"""You are a compliance expert analyzing regulatory requirements.

REGULATORY REQUIREMENT:
Title: {regulation_title}
Text: {regulation_text}

AVAILABLE CONTROLS IN THIS BATCH:
{control_list}

TASK:
For each control in this batch, rate its relevance to the regulatory requirement:
- Score 1-10 where 10 = ESSENTIAL (must have)
- Score 7-9 = HIGHLY RELEVANT (strongly recommended)
- Score 4-6 = RELEVANT (somewhat related)
- Score 1-3 = MINOR (loosely related)

IMPORTANT: Only include controls with score >= 4

RETURN FORMAT (only controls with score 4+):
L1C-001:8
L1C-045:7
L1C-102:5

If no controls score 4+, return: NONE"""

        try:
            response = client.messages.create(
                model="gpt-4o-mini",
                max_tokens=500,
                temperature=0.2,
                messages=[
                    {
                        "role": "user",
                        "content": prompt
                    }
                ]
            )
            
            response_text = response.content[0].text.strip()
            
            # Parse response: extract control:score pairs
            for line in response_text.split('\n'):
                line = line.strip()
                if ':' in line and line != 'NONE':
                    parts = line.split(':')
                    if len(parts) == 2:
                        ctrl_id = parts[0].strip().upper()
                        try:
                            score = float(parts[1].strip())
                            if ctrl_id in control_ids and 4 <= score <= 10:
                                all_controls_with_scores[ctrl_id] = score
                        except ValueError:
                            pass
            
            print(f"  Batch {batch_num + 1}/{len(control_batches)}: Found {len([s for s in all_controls_with_scores.values() if s >= 4])} controls")
        
        except Exception as e:
            print(f"  ⚠ Error processing batch {batch_num + 1}: {str(e)}")
    
    # Sort by confidence score and select top K
    if not all_controls_with_scores:
        return [], {}
    
    sorted_controls = sorted(all_controls_with_scores.items(), key=lambda x: x[1], reverse=True)
    top_controls = sorted_controls[:top_k]
    
    final_controls = [ctrl_id for ctrl_id, score in top_controls]
    final_scores = {ctrl_id: score for ctrl_id, score in top_controls}
    
    return final_controls, final_scores

def rank_and_select_top_controls(regulation_title, regulation_text, 
                                candidate_controls, control_catalog, top_k=10):
    """
    Use LLM to rank candidate controls and select top 10.
    """
    
    if not candidate_controls:
        return []
    
    if len(candidate_controls) <= top_k:
        return candidate_controls
    
    # Create list of candidates
    candidate_list = ""
    for i, ctrl_id in enumerate(sorted(candidate_controls)[:50], 1):  # Show top 50 for ranking
        candidate_list += f"{i}. {control_catalog[ctrl_id]}\n"
    
    prompt = f"""You are a compliance expert ranking security controls.

REGULATORY REQUIREMENT:
Title: {regulation_title}
Text: {regulation_text}

CANDIDATE CONTROLS (in no particular order):
{candidate_list}

TASK: Rank these controls by IMPORTANCE to this regulation.
Select the TOP {top_k} MOST IMPORTANT controls.

Return ONLY the {top_k} control IDs in order of importance (most important first):"""

    try:
        response = client.messages.create(
            model="gpt-4o-mini",
            max_tokens=300,
            temperature=0.1,
            messages=[
                {
                    "role": "user",
                    "content": prompt
                }
            ]
        )
        
        response_text = response.content[0].text.strip()
        
        # Parse response and extract top controls
        ranked_controls = []
        for line in response_text.split('\n'):
            line = line.strip().upper()
            if line:
                # Extract control ID (first token)
                ctrl_id = line.split()[0] if line else None
                if ctrl_id and ctrl_id in candidate_controls:
                    ranked_controls.append(ctrl_id)
                    if len(ranked_controls) >= top_k:
                        break
        
        return ranked_controls[:top_k]
    
    except Exception as e:
        print(f"  ⚠ Error ranking controls: {str(e)}")
        return candidate_controls[:top_k]
