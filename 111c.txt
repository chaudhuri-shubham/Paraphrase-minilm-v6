import pandas as pd

def load(path):
    return pd.read_excel(path) if path.endswith(('.xlsx', '.xls')) else pd.read_csv(path)

reg_df    = load(REG_FILE)
ctrl_df   = load(CONTROLS_FILE)
golden_df = load(GOLDEN_FILE)

# Ensure all IDs are strings
reg_df[REG_ID_COL]       = reg_df[REG_ID_COL].astype(str)
ctrl_df[CTRL_ID_COL]     = ctrl_df[CTRL_ID_COL].astype(str)
golden_df[GS_REG_COL]    = golden_df[GS_REG_COL].astype(str)
golden_df[GS_CTRL_COL]   = golden_df[GS_CTRL_COL].astype(str)

# Compact title-only list for Call 1 (low token cost)
all_titles = "\n".join(
    f"{r[CTRL_ID_COL]}: {r[CTRL_TITLE_COL]}"
    for _, r in ctrl_df.iterrows()
)

# Full description lookup for Call 2
ctrl_desc_map = {
    r[CTRL_ID_COL]: f"{r[CTRL_TITLE_COL]} — {r[CTRL_DESC_COL]}"
    for _, r in ctrl_df.iterrows()
}

print(f"✅ {len(reg_df)} regulatory texts | {len(ctrl_df)} controls loaded")
Cell 5 — GPT helper + two mapping calls
import json, time
from openai import OpenAI

client = OpenAI(api_key=API_KEY)

def call_gpt(system, user, retries=3):
    """Single GPT call with retry. Returns parsed JSON or None."""
    for attempt in range(retries):
        try:
            resp = client.chat.completions.create(
                model=MODEL,
                temperature=0,
                max_tokens=600,
                messages=[
                    {"role": "system", "content": system},
                    {"role": "user",   "content": user}
                ]
            )
            raw = resp.choices[0].message.content.strip()
            raw = raw.lstrip("```json").lstrip("```").rstrip("```").strip()
            return json.loads(raw)
        except json.JSONDecodeError:
            print(f"    ⚠ JSON parse error on attempt {attempt+1}, retrying...")
            time.sleep(2)
        except Exception as e:
            print(f"    ⚠ API error on attempt {attempt+1}: {e}")
            time.sleep(5)
    return None


def get_shortlist(reg_id, reg_text):
    """
    CALL 1 — Send all control titles (no descriptions) in one shot.
    Ask GPT to return ~SHORTLIST_SIZE candidate IDs broadly.
    Tokens: very low (titles only, no descriptions).
    """
    system = f"""You are a compliance expert. 
Given a regulatory text, return the IDs of the {SHORTLIST_SIZE} most relevant controls from the list.
Cast a wide net — include direct AND indirect matches.
Return ONLY a JSON array of ID strings: ["CTL-001", "CTL-005", ...]
No prose, no markdown."""

    user = f"Regulatory text (ID: {reg_id}):\n{reg_text}\n\nAll controls (ID: Title):\n{all_titles}"

    result = call_gpt(system, user)
    if not isinstance(result, list):
        print(f"  ⚠ Shortlist failed for {reg_id}, skipping.")
        return []

    # Keep only valid IDs that exist in our controls file
    valid = [str(i) for i in result if str(i) in ctrl_desc_map]
    return valid


def verify_and_score(reg_id, reg_text, shortlist_ids):
    """
    CALL 2 — Send only shortlisted controls with FULL descriptions.
    Ask GPT to score each and return top MAX_FINAL matches.
    Tokens: low (only 20–30 controls with descriptions).
    """
    shortlist_text = "\n".join(
        f"[{cid}] {ctrl_desc_map[cid]}"
        for cid in shortlist_ids
    )

    system = f"""You are a compliance expert. Score each control's relevance to the regulatory text.
Confidence 0–10:
  9–10 = direct, explicit match
  6–8  = strong thematic match
  4–5  = indirect but plausible
  0–3  = not relevant — exclude

Rules:
- Return a maximum of {MAX_FINAL} controls (the most relevant ones)
- Only include controls with confidence >= {MIN_CONF}
- Return ONLY a JSON array sorted by confidence descending:
  [{{"control_id":"...","confidence":8,"reason":"one short sentence"}}]
- Return [] if nothing qualifies. No prose, no markdown."""

    user = f"Regulatory text (ID: {reg_id}):\n{reg_text}\n\nCandidate controls:\n{shortlist_text}"

    result = call_gpt(system, user)
    if not isinstance(result, list):
        return []

    verified = []
    for m in result:
        if (isinstance(m, dict)
                and str(m.get("control_id", "")) in ctrl_desc_map
                and m.get("confidence", 0) >= MIN_CONF):
            verified.append({
                "reg_id"     : reg_id,
                "control_id" : str(m["control_id"]),
                "confidence" : m["confidence"],
                "reason"     : m.get("reason", "")
            })

    # Hard cap at MAX_FINAL
    return sorted(verified, key=lambda x: x["confidence"], reverse=True)[:MAX_FINAL]
Cell 6 — Run mapping
detailed_rows = []

for _, row in reg_df.iterrows():
    reg_id   = row[REG_ID_COL]
    reg_text = str(row[REG_TEXT_COL])

    print(f"\n{'═'*55}")
    print(f"  {reg_id}")
    print(f"{'═'*55}")

    # Call 1 — broad shortlist from titles only
    print(f"  Call 1: Scanning all {len(ctrl_df)} control titles...")
    shortlist = get_shortlist(reg_id, reg_text)
    print(f"  → {len(shortlist)} candidates: {shortlist}")

    if not shortlist:
        continue

    # Call 2 — score shortlist with full descriptions, keep top MAX_FINAL
    print(f"  Call 2: Scoring {len(shortlist)} candidates with full descriptions...")
    verified = verify_and_score(reg_id, reg_text, shortlist)
    print(f"  → {len(verified)} final matches (max {MAX_FINAL}, conf ≥ {MIN_CONF}):")
    for m in verified:
        print(f"     [{m['confidence']}] {m['control_id']} — {m['reason'][:70]}")

    detailed_rows.extend(verified)
    time.sleep(0.5)

print(f"\n\n✅ Mapping complete — {len(detailed_rows)} total matches")
Cell 7 — Build output dataframes
# Detailed table — one row per match with confidence and reason
detailed_df = (
    pd.DataFrame(detailed_rows)
    .merge(ctrl_df[[CTRL_ID_COL, CTRL_TITLE_COL]], on=CTRL_ID_COL, how="left")
    .sort_values(["reg_id", "confidence"], ascending=[True, False])
    .reset_index(drop=True)
)

# ── FINAL OUTPUT: reg_id | mapped_control_ids ─────────────────────────────
final_df = (
    detailed_df
    .groupby("reg_id")["control_id"]
    .apply(lambda ids: ", ".join(sorted(ids)))
    .reset_index()
    .rename(columns={"control_id": "mapped_control_ids"})
)

print("FINAL OUTPUT:")
print(final_df.to_string(index=False))
Cell 8 — Validate against golden source
pred_pairs = set(zip(detailed_df["reg_id"], detailed_df["control_id"]))
gold_pairs = set(zip(golden_df[GS_REG_COL], golden_df[GS_CTRL_COL]))

tp        = pred_pairs & gold_pairs
fp        = pred_pairs - gold_pairs
fn        = gold_pairs - pred_pairs
precision = len(tp) / len(pred_pairs) if pred_pairs else 0
recall    = len(tp) / len(gold_pairs) if gold_pairs else 0
f1        = 2*precision*recall / (precision+recall) if (precision+recall) > 0 else 0

print(f"{'═'*40}")
print(f"  VALIDATION vs GOLDEN SOURCE")
print(f"{'═'*40}")
print(f"  Predicted    : {len(pred_pairs)}")
print(f"  Golden       : {len(gold_pairs)}")
print(f"  True  Pos    : {len(tp)}")
print(f"  False Pos    : {len(fp)}")
print(f"  Missed (FN)  : {len(fn)}")
print(f"  ──────────────────────────────────")
print(f"  Precision    : {precision:.1%}")
print(f"  Recall       : {recall:.1%}")
print(f"  F1 Score     : {f1:.1%}")
if fn:
    print(f"  Missed pairs : {sorted(fn)}")
