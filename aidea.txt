# --- Core Financial Data and Configuration (Adjusted for INR) ---

GOAL_NAME = "First Home Down Payment Fund"
GOAL_TARGET = 5000000.00 # 50 Lakhs INR
GOAL_DEADLINE_YEAR = 2026
RISK_THRESHOLD_PERCENT = 0.90 
RISKY_AMOUNT_DEFAULT = 650000.00 # Default amount used to trigger the Nudge (6.5 Lakhs)

# Initial balances
INITIAL_BALANCES = {
    'Joint Checking': 4550000.00, # 45.5 Lakhs (Goal Balance)
    'Discretionary Savings': 1000000.00, # 10 Lakhs
    'Partner\'s Account': 50000.00, # 50K (Simulated Partner Contribution)
}

# Simulated Transaction Data for Insight Generation (Last 30 Days) - Adjusted for INR
EXPENSE_DATA = {
    'Category': ['Groceries', 'Groceries', 'Dining Out', 'Dining Out', 'Rent/Mortgage', 'Utilities', 'Subscription', 'Travel', 'Groceries', 'Shopping', 'Investment', 'Investment'],

}

# --- Logic Functions for the 5 Layers ---

# 3. Nudge Logic (Protect)
def check_transaction_risk(source_account, amount):
    """Checks if the transaction derails the goal."""
    if source_account != 'Joint Checking':
        return False, st.session_state.balances['Joint Checking'], "None"

    current_goal_balance = st.session_state.balances['Joint Checking']
    balance_after = current_goal_balance - amount
    
    is_risky = balance_after < (GOAL_TARGET * RISK_THRESHOLD_PERCENT)
    
    # Simple simulation for delay: 9 months for a 6.5 Lakhs hit.
    delay_factor = amount / RISKY_AMOUNT_DEFAULT
    simulated_delay = f"{round(delay_factor * 9, 0)} months" if is_risky else "None"
        
    return is_risky, balance_after, simulated_delay

def execute_transfer(source, target, amount):
    """Executes the transaction and updates balances."""
    if st.session_state.balances[source] >= amount:
        st.session_state.balances[source] -= amount
        st.session_state.status_message = f"✅ Success! Transferred ₹{amount:,.2f} from {source}."
    else:
        st.session_state.status_message = "❌ Error: Insufficient funds."

def handle_reallocation(risky_amount):
    """Handles the Nudge-recommended reallocation."""
    source_risky = 'Joint Checking'
    source_safe = 'Discretionary Savings'
    
    if st.session_state.balances[source_safe] >= risky_amount:
        st.session_state.balances[source_safe] -= risky_amount
        st.session_state.balances[source_risky] += risky_amount
        
        st.session_state.status_message = f"""
        ✨ **REALLOCATION SUCCESS!** ✨ The ₹{risky_amount:,.2f} purchase was completed using **{source_safe}**.
        Your **{GOAL_NAME}** fund is protected!
        """
        st.session_state.is_risky_transaction = False
    else:
        st.session_state.status_message = "❌ Error: Insufficient funds in Discretionary Savings."

# 1. Transaction Insights Logic (Analyze)
def generate_financial_insight(user_query, expense_df):
    """Generates hardcoded, simulated financial insights."""
    user_query = user_query.lower()
    
    # Calculate monthly summary for contextual data
    monthly_summary = expense_df.groupby('Category')['Amount'].sum().reset_index()
    
    # Logic to provide tailored insights based on query
    if "groceries" in user_query or "food" in user_query:
        # Groceries Insight: Assume typical monthly spend is 25,000 INR
        try:
            groceries_total = monthly_summary[monthly_summary['Category'] == 'Groceries']['Amount'].iloc[0]
        except:
            groceries_total = 0.00
            
        overage = max(0, groceries_total - 25000.00) 
        
        return (
            f"**Actionable Groceries Insight:** You spent **₹{groceries_total:,.2f}** this month. "
            f"Your average is usually ₹25,000. Cutting this **₹{overage:,.2f}** overage could be put directly "
            f"into your '{GOAL_NAME}' fund! **Recommendation:** Plan meals and commit to a 'no-spend' cooking day weekly."
        )
    elif "dining" in user_query or "restaurant" in user_query:
        # Dining Out Insight: Assume cutting 2-3 trips saves 10,000 INR
        try:
            dining_total = monthly_summary[monthly_summary['Category'] == 'Dining Out']['Amount'].iloc[0]
        except:
            dining_total = 0.00

        savings = 10000.00
        
        return (
            f"**Dining Out Optimization:** Your **₹{dining_total:,.2f}** on Dining Out is a high priority area. "
            f"By cutting 2-3 restaurant trips, you could save about **₹{savings:,.2f}** over the next month, accelerating your home goal significantly!"
        )
    else:
        # Default response
        return (
            "Simulated Insight: I'm the **Personalized Education Engine**. I can analyze your spending to find ways to boost your savings. "
            "Try asking me specifically about **'groceries'** or **'dining out'** for a tailor-made plan!"
        )

# 4. Product Recommendation Logic (Promote)
def get_product_recommendation():
    """Generates product recommendation based on user goal and risk."""
    if st.session_state.is_partner_added:
        return "🤝 **HSBC Joint Fixed-Rate Deposit (JFRD):** Offers a high interest rate, perfect for accelerating your shared goal. Since you are saving collaboratively, this joint account maximizes returns."
    
    current_goal_balance = st.session_state.balances['Joint Checking']
    if current_goal_balance > 4500000: # Over 45 Lakhs (Close to target)
        return "📈 **HSBC Low-Risk Capital Bond:** Since you are very close to your target, consider putting the last few lakhs into a short-term, low-risk bond to earn maximum safe interest before withdrawal."
    else:
        return "🔒 **HSBC Flexi-Term RD (Recurring Deposit):** Start a separate, automated monthly RD to systematically fund your Discretionary Savings. This protects your main goal."

# 5. Goal Reporting Logic (Report)
def generate_progress_report(current_balance, target):
    """Generates a detailed progress report summary."""
    time_remaining_months = (GOAL_DEADLINE_YEAR - 2025) * 12 # Simple simulation: 12 months left
    
    if target == 0: return "Set a goal target first!"
    
    required_monthly = (target - current_balance) / time_remaining_months if time_remaining_months > 0 and current_balance < target else 0
    
    if current_balance >= target:
        status = "Goal Achieved!"
    elif required_monthly > 100000: # Over 1 Lakh per month is high risk
        status = "⚠️ **High-Risk:** Monthly contribution needed is high. You must increase savings or extend the timeline."
    else:
        status = "🟢 **On Track:** Maintain the current contribution pace."
        
    report = f"""
    - **Goal Status:** {status}
    - **Current Progress:** {(current_balance / target) * 100:.1f}%
    - **Estimated Finish Date:** Q4 {GOAL_DEADLINE_YEAR}
    - **Required Monthly Contribution:** ₹{required_monthly:,.2f}
    """
    return report
