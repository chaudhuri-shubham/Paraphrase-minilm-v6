def validate_controls_with_gpt4(regulation_id, regulation_text, controls_to_validate, 
                                control_descriptions):
    """
    Use GPT-4 to validate if each control is truly relevant to the regulation.
    
    Key points:
    - Be STRICT: Only YES if control is directly and clearly relevant
    - Consider implementation, not just mention
    - Conservative approach: when in doubt, say NO
    """
    
    if not controls_to_validate or len(controls_to_validate) == 0:
        return set()
    
    # Build control descriptions list
    control_info = ""
    for ctrl in sorted(controls_to_validate):
        desc = control_descriptions.get(ctrl, "No description available")
        control_info += f"- {ctrl}: {desc}\n"
    
    # STRICT validation prompt for GPT-4
    system_prompt = """You are a strict compliance and security controls expert. 
Your task is to validate if specific controls are relevant to implementing or supporting 
a given regulatory requirement.

Be CONSERVATIVE and STRICT:
- Only answer YES if the control is DIRECTLY and CLEARLY related to the regulation
- Consider actual implementation of the control, not just semantic similarity
- When in doubt, answer NO - we prefer fewer correct controls than many incorrect ones
- A control that "could help" is not relevant - it must be NECESSARY or HIGHLY RELEVANT

Your response format:
For EACH control, respond with: CONTROL_ID:YES or CONTROL_ID:NO
List only the format above, no explanations."""

    user_prompt = f"""Regulatory Requirement:
"{regulation_text}"

Regulation ID: {regulation_id}

Below are controls that were initially mapped to this requirement.
Evaluate EACH control strictly:

Controls to validate:
{control_info}

For each control above, respond with: CONTROL_ID:YES or CONTROL_ID:NO
Remember: Be STRICT and CONSERVATIVE. Only YES if directly and clearly relevant.

Response (format: CONTROL_ID:YES/NO for each control):"""

    try:
        response = client.messages.create(
            model="gpt-4-turbo",
            max_tokens=500,
            temperature=0.2,  # Lower temperature = more consistent, stricter responses
            messages=[
                {
                    "role": "user",
                    "content": user_prompt
                }
            ],
            system=system_prompt
        )
        
        response_text = response.content[0].text.strip()
        
        # Parse response
        validated_controls = set()
        
        for line in response_text.split('\n'):
            line = line.strip()
            if ':' in line and not line.startswith('Note') and not line.startswith('Remember'):
                parts = line.split(':')
                if len(parts) >= 2:
                    ctrl_id = parts[0].strip().upper()
                    answer = parts[1].strip().upper()
                    
                    # Check if this is a valid control we're validating
                    if ctrl_id in controls_to_validate and 'YES' in answer:
                        validated_controls.add(ctrl_id)
        
        return validated_controls
    
    except Exception as e:
        print(f"  ⚠ Error in GPT-4 validation for {regulation_id}: {str(e)}")
        return controls_to_validate

def apply_strict_similarity_filter(controls, similarity_dict, 
                                  min_connections=2, similarity_threshold=0.75):
    """
    Apply STRICT similarity filtering.
    
    Keep control ONLY if:
    1. Has at least 2 strong connections (similarity >= 0.75) to other controls
    2. Average similarity to other controls is high (>= 0.65)
    
    This removes:
    - Isolated controls
    - Weak/uncertain controls
    - Controls that don't fit the semantic pattern
    """
    
    if len(controls) <= 1:
        return controls
    
    filtered = set()
    filter_log = {}
    
    for ctrl in controls:
        if ctrl not in similarity_dict:
            # Not in similarity matrix = disqualify
            filter_log[ctrl] = 'Not in similarity matrix'
            continue
        
        # Count strong connections
        strong_connections = []
        all_similarities = []
        
        for other_ctrl in controls:
            if ctrl != other_ctrl and other_ctrl in similarity_dict[ctrl]:
                sim = similarity_dict[ctrl][other_ctrl]
                all_similarities.append(sim)
                
                if sim >= similarity_threshold:
                    strong_connections.append(sim)
        
        # Keep ONLY if has enough strong connections
        if len(strong_connections) >= min_connections:
            filtered.add(ctrl)
            filter_log[ctrl] = f'KEPT: {len(strong_connections)} strong connections'
        elif all_similarities:
            # Fallback: Check average similarity
            avg_sim = np.mean(all_similarities)
            if avg_sim >= 0.65 and len(strong_connections) >= 1:
                filtered.add(ctrl)
                filter_log[ctrl] = f'KEPT: Avg similarity {avg_sim:.3f}, {len(strong_connections)} strong'
            else:
                filter_log[ctrl] = f'REMOVED: Avg similarity {avg_sim:.3f}, only {len(strong_connections)} strong'
        else:
            filter_log[ctrl] = 'REMOVED: No similarities to other controls'
    
    # Ensure at least 1 control remains
    if len(filtered) == 0 and len(controls) > 0:
        # Keep the control with best overall connections
        best_ctrl = None
        best_score = -1
        
        for ctrl in controls:
            if ctrl not in similarity_dict:
                continue
            
            sims = [similarity_dict[ctrl].get(other, 0) for other in controls if other != ctrl]
            if sims:
                avg_sim = np.mean(sims)
                if avg_sim > best_score:
                    best_score = avg_sim
                    best_ctrl = ctrl
        
        if best_ctrl:
            filtered.add(best_ctrl)
            filter_log[best_ctrl] = f'KEPT (fallback): Best average similarity {best_score:.3f}'
        else:
            filtered = controls
    
    return filtered, filter_log

def calculate_confidence_score(control_id, regulation_controls, similarity_dict):
    """
    Calculate confidence score for each control (0-1 scale).
    
    Factors:
    - Connection strength: Max similarity to any other control
    - Network density: Number of strong connections (>0.65)
    - Average similarity: How well it fits with the group
    """
    
    if control_id not in similarity_dict:
        return 0.0
    
    # Factor 1: Max similarity (0-0.4)
    max_sim = 0.0
    for other_ctrl in regulation_controls:
        if control_id != other_ctrl and other_ctrl in similarity_dict[control_id]:
            max_sim = max(max_sim, similarity_dict[control_id][other_ctrl])
    max_sim_score = max_sim * 0.4
    
    # Factor 2: Strong connections count (0-0.3)
    strong_count = 0
    for other_ctrl in regulation_controls:
        if control_id != other_ctrl and other_ctrl in similarity_dict[control_id]:
            if similarity_dict[control_id][other_ctrl] >= 0.65:
                strong_count += 1
    
    max_possible_connections = max(len(regulation_controls) - 1, 1)
    strong_connection_score = (strong_count / max_possible_connections) * 0.3
    
    # Factor 3: Average similarity (0-0.3)
    similarities = []
    for other_ctrl in regulation_controls:
        if control_id != other_ctrl and other_ctrl in similarity_dict[control_id]:
            similarities.append(similarity_dict[control_id][other_ctrl])
    
    if similarities:
        avg_sim_score = np.mean(similarities) * 0.3
    else:
        avg_sim_score = 0.0
    
    total_confidence = max_sim_score + strong_connection_score + avg_sim_score
    
    return total_confidence
