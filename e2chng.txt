import os
import fitz               # PDF - PyMuPDF
import extract_msg        # MSG reader
from bs4 import BeautifulSoup
from docx import Document


# -------------------------------------------------------
# PDF CHECK
# -------------------------------------------------------
def check_pdf_content(file_path):
    """
    Returns 1 ONLY IF:
      - PDF has embedded images
      - AND text length == 0
    Else returns 0
    """
    try:
        doc = fitz.open(file_path)
    except Exception as e:
        print("PDF open error:", e)
        return 0

    has_image = False
    text_len = 0

    for page in doc:
        # Image check
        images = page.get_images(full=True)
        if images:
            has_image = True

        # Text check
        txt = page.get_text("text").strip()
        if txt:
            text_len += len(txt)

    # return 1 only when image exists & no text
    if has_image and text_len == 0:
        return 1
    return 0



# -------------------------------------------------------
# MSG CHECK
# -------------------------------------------------------
def check_msg_content(file_path):
    """
    Returns 1 ONLY IF:
      - MSG has inline/attached images
      - AND no meaningful text in body
    """
    try:
        msg = extract_msg.Message(file_path)
        msg.extract()
    except Exception as e:
        print("MSG open error:", e)
        return 0

    image_exts = (".jpg", ".jpeg", ".png", ".gif", ".bmp", ".tiff", ".webp")
    has_image = False

    # --- Check attachments (inline or normal) ---
    for att in msg.attachments:
        name = (att.longFilename or att.shortFilename or "").lower()
        if name.endswith(image_exts):
            has_image = True

    # --- Check text from body & html ---
    text_content = ""

    if msg.body:
        text_content += msg.body.strip()

    if msg.htmlBody:
        soup = BeautifulSoup(msg.htmlBody, "html.parser")
        text_content += soup.get_text(separator=" ").strip()

    text_len = len(text_content.strip())

    # return 1 only when image exists & no text
    if has_image and text_len == 0:
        return 1
    return 0



# -------------------------------------------------------
# DOCX CHECK
# -------------------------------------------------------
def check_docx_content(file_path):
    """
    Returns 1 ONLY IF:
      - DOCX has embedded images
      - AND no text found in document
    """
    try:
        doc = Document(file_path)
    except Exception as e:
        print("DOCX open error:", e)
        return 0

    has_image = False
    text_len = 0

    # ---- image check: embedded images ----
    for rel in doc.part._rels.values():
        try:
            part = rel.target_part
            if hasattr(part, 'content_type'):
                if part.content_type.startswith("image/"):
                    has_image = True
        except:
            pass

    # ---- text check ----
    for para in doc.paragraphs:
        if para.text.strip():
            text_len += len(para.text.strip())

    for table in doc.tables:
        for row in table.rows:
            for cell in row.cells:
                if cell.text.strip():
                    text_len += len(cell.text.strip())

    # return 1 only when image exists & no text
    if has_image and text_len == 0:
        return 1
    return 0



# -------------------------------------------------------
# COMBINED MASTER FUNCTION
# -------------------------------------------------------
def check_file_content(file_path):
    """
    Detect file type and call the correct function.
    Returns 1 only if:
      - image present
      - no text present
    Valid for: PDF, DOCX, MSG
    """
    ext = os.path.splitext(file_path)[1].lower()

    if ext == ".pdf":
        return check_pdf_content(file_path)
    elif ext == ".msg":
        return check_msg_content(file_path)
    elif ext == ".docx":
        return check_docx_content(file_path)
    else:
        print("Unsupported file format:", ext)
        return 0

# -------------------------------------------------------
# PDF CHECK — PyMuPDF
# -------------------------------------------------------
def check_pdf_content(file_path):
    """
    Returns 1 ONLY IF:
      - PDF has embedded images AND no extractable text
    """

        doc = fitz.open(file_path)
    except Exception as e:
        print("PDF open error:", e)
        return 0

    has_image = False
    text_length = 0

    for page in doc:
        # --- IMAGE CHECK ---
        images = page.get_images(full=True)
        if images:
            has_image = True

        # --- TEXT CHECK ---
        text = page.get_text("text").strip()
        if text:
            text_length += len(text)

    if has_image and text_length == 0:
        return 1
    return 0



# -------------------------------------------------------
# MSG CHECK — extract_msg (NO extract(), NO BeautifulSoup)
# -------------------------------------------------------
def check_msg_content(file_path):
    """
    Returns 1 ONLY IF:
      - MSG contains images (inline or attachment)
      - AND email has no text (plain or html)
    """

    try:
        msg = extract_msg.Message(file_path)
    except Exception as e:
        print("MSG open error:", e)
        return 0

    has_image = False
    text_length = 0

    # --------- TEXT CHECK (no HTML parsing) ---------
    if msg.body:
        text_length += len(msg.body.strip())

    if msg.htmlBody:
        # treat HTML as raw text - no BeautifulSoup
        raw_html = msg.htmlBody.strip()
        # remove < > to avoid counting tags as text
        cleaned = raw_html.replace("<", " ").replace(">", " ").strip()
        text_length += len(cleaned)

    # --------- IMAGE CHECK (attachments + inline) ---------
    image_exts = (".jpg", ".jpeg", ".png", ".gif", ".bmp", ".tiff", ".webp")

    for att in msg.attachments:
        name = (att.longFilename or att.shortFilename or "").lower()
        if name.endswith(image_exts):
            has_image = True

    if has_image and text_length == 0:
        return 1
    return 0

import os
import fitz                     # for PDF
from docx import Document       # for DOCX
import extract_msg              # for MSG


# ---------------------------
# Custom Exception
# ---------------------------
class OnlyImageNoTextError(Exception):
    """Raised when a file contains only images and no text."""
    pass


# ---------------------------
# Helper: Check PDF
# ---------------------------
def _pdf_check(file_path):
    has_text = 0
    has_image = 0

    doc = fitz.open(file_path)

    for page in doc:
        # check text
        if page.get_text(strip=True):
            has_text = 1

        # check images
        if page.get_images(full=True):
            has_image = 1

    return has_text, has_image


# ---------------------------
# Helper: Check DOCX
# ---------------------------
def _docx_check(file_path):
    has_text = 0
    has_image = 0

    doc = Document(file_path)

    # text check
    full_text = "\n".join([para.text for para in doc.paragraphs])
    if full_text.strip():
        has_text = 1

    # images check
    for rel in doc.part._rels:
        target = doc.part._rels[rel].target_ref
        if any(target.lower().endswith(ext) for ext in
               (".png", ".jpg", ".jpeg", ".gif", ".bmp", ".tiff", ".webp")):
            has_image = 1

    return has_text, has_image


# ---------------------------
# Helper: Check MSG
# ---------------------------
def _msg_check(file_path):
    has_text = 0
    has_image = 0

    msg = extract_msg.Message(file_path)

    # text check
    body = msg.body or ""
    if body.strip():
        has_text = 1

    # attachments (MSG stores embedded images as attachments)
    image_exts = (".png", ".jpg", ".jpeg", ".gif", ".bmp", ".tiff", ".webp")

    for att in msg.attachments:
        name = (att.longFilename or att.shortFilename or "").lower()
        if name.endswith(image_exts):
            has_image = 1

    return has_text, has_image


# ---------------------------
# MAIN UNIVERSAL CHECKER FUNCTION
# ---------------------------
def analyze_file_for_image_text(file_path):
    """
    Call this inside your EXCEPTION block.
    Checks whether file has text, images, or only images.
    Raises OnlyImageNoTextError when only images present.
    """

    ext = os.path.splitext(file_path)[-1].lower()

    if ext == ".pdf":
        has_text, has_image = _pdf_check(file_path)

    elif ext == ".docx":
        has_text, has_image = _docx_check(file_path)

    elif ext == ".msg":
        has_text, has_image = _msg_check(file_path)

    else:
        raise ValueError("Unsupported file type for fallback check.")

    # --------- CUSTOM EXCEPTION ---------
    if has_image == 1 and has_text == 0:
        raise OnlyImageNoTextError("File contains only images and no text.")

    # normal return
    return {
        "has_image": has_image,
        "has_text": has_text
    }

from paddleocr import PaddleOCR
import os

ocr = PaddleOCR(use_angle_cls=True, lang='en')

def ocr_image_paddle(file_name):
    if not os.path.exists(file_name):
        return 0, ""

    result = ocr.ocr(file_name, cls=True)
    extracted = []

    if result and result[0]:
        for line in result[0]:
            extracted.append(line[1][0])

    if extracted:
        return 1, " ".join(extracted)
    else:
        return 0, ""
