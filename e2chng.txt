import pandas as pd

# Read input files
df1 = pd.read_excel("file1.xlsx")   # original data
df2 = pd.read_excel("file2.xlsx")   # mapping table

# Keep original row identity
df1["_row_id"] = df1.index

# Split values ONLY for internal matching
tmp = (
    df1
    .assign(
        code=df1["col1"].str.split("|"),
        text=df1["col2"].str.split("|")
    )
    .explode("code")
    .explode("text")
)

# Map with class table
tmp = tmp.merge(
    df2,
    on=["code", "text"],
    how="left"
)

# Identify rows having at least one MAJOR mapping
major_rows = tmp.loc[
    tmp["class_type"].eq("major"),
    "_row_id"
].unique()

# Filter original dataframe (structure unchanged)
final_df = (
    df1[df1["_row_id"].isin(major_rows)]
    .drop(columns="_row_id")
)

print(final_df)
