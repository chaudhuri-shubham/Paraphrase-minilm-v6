def generate_financial_insight(user_query, expense_df):
    """Generates insights based on multi-month data comparison."""
    user_query = user_query.lower()
    
    current_month = 'Nov-25'
    prev_months = ['Sep-25', 'Oct-25']
    
    df_current = expense_df[expense_df['Month'] == current_month]
    df_prev = expense_df[df_expenses['Month'].isin(prev_months)]

    current_spend = df_current.groupby('Category')['Amount'].sum()
    prev_avg_spend = df_prev.groupby('Category')['Amount'].sum() / len(prev_months)
    
    # --- Groceries Logic ---
    current_groceries = current_spend.get('Groceries', 0)
    avg_groceries = prev_avg_spend.get('Groceries', 0)
    groceries_diff = current_groceries - avg_groceries

    # --- Healthcare Logic ---
    current_healthcare = current_spend.get('Healthcare', 0)
    avg_healthcare = prev_avg_spend.get('Healthcare', 0)
    healthcare_increase_pct = (current_healthcare / avg_healthcare) * 100 if avg_healthcare > 0 else 0

    # --- Intelligent Insights based on Variance ---
    
    if "groceries" in user_query or "food" in user_query:
        if groceries_diff > 10000: # Significant increase (high spike in Nov)
             return (
                f"**ALERT: Grocery Trend Spike!** Your spend this month (₹{current_groceries:,.0f}) is **₹{groceries_diff:,.0f}** higher than your 2-month average (₹{avg_groceries:,.0f}). "
                f"This sharp spike suggests stock-up buying or unnecessary premium purchases. **Action:** Audit November's shopping list. You can save ₹{groceries_diff:,.0f} by reverting to average spend."
            )
        elif groceries_diff < -5000: # Significant decrease (good!)
            return (
                f"**WELL DONE!** Your groceries spend is **₹{abs(groceries_diff):,.0f}** below average. Keep this habit up, and transfer that saving to your goal fund immediately to accelerate your progress!"
            )
        else:
             return (
                f"Your groceries spend (₹{current_groceries:,.0f}) is on par with your average. If you could reduce this by just ₹5,000, you would save an extra ₹60,000 annually for your down payment."
            )

    elif "healthcare" in user_query or "health" in user_query:
        if healthcare_increase_pct > 150: # More than double the average
             return (
                f"**CRITICAL INSIGHT: Increasing Health Spend!** Your Healthcare expenses are up {healthcare_increase_pct:.0f}% from the 2-month average (₹{avg_healthcare:,.0f}). "
                f"This non-discretionary increase should be tracked. **Action:** Review your insurance coverage and consider setting aside a small, fixed monthly amount into a dedicated 'Health Reserve' fund."
            )
        else:
             return (
                f"Your healthcare spending remains manageable. Ensure you are maximizing benefits from your current health insurance policy."
            )

    elif "transport" in user_query or "travel" in user_query:
        # Check if Travel is high but Transportation is low (suggesting one big trip over commuting)
        current_travel = current_spend.get('Travel & Vacations', 0)
        current_transport = current_spend.get('Transportation', 0)
        
        if current_travel > 4000 and current_transport < 1000:
            return (
                f"**TRAVEL DISCIPLINE:** We see a ₹{current_travel:,.0f} Travel expense, but your regular Transportation costs are near zero. This indicates a high-value, specific trip. **Recommendation:** Since you have saved on daily commutes, ensure this travel expense was budgeted, and transfer any unexpected leftover savings from this category to your home goal."
            )
        else:
            return (
                f"Transportation and travel costs are consistent. Review public transport passes or carpooling options if you want to find marginal savings."
            )
            
    elif "utility" in user_query or "utilities" in user_query:
        current_utilities = current_spend.get('Utilities', 0)
        avg_utilities = prev_avg_spend.get('Utilities', 0)

        if current_utilities > avg_utilities * 1.15: # 15% increase
            return (
                f"**UTILITY SPIKE ALERT:** Your Utilities expense (₹{current_utilities:,.0f}) is 15% higher than your average. This could indicate seasonal change or a leak. **Action:** Review usage patterns immediately to identify and correct the cause, protecting your monthly savings flow."
            )
        else:
             return (
                f"Your utility costs are stable. Good job maintaining efficiency!"
            )
    
    else:
        return (
            "I've analyzed your 3-month trend. Your **Groceries** and **Healthcare** show the biggest change this month. Try asking: **'What about groceries?'** or **'Tell me about healthcare spending.'**"
        )
