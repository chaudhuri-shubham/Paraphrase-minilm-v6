/* Minimal RFC4180-ish CSV parser: handles quoted fields, embedded commas,
   escaped "" quotes, and \r\n or \n line endings. No external library needed. */
function parseCSV(text){
  const rows = [];
  let row = [], field = '', inQuotes = false;
  for (let i = 0; i < text.length; i++){
    const c = text[i];
    if (inQuotes){
      if (c === '"'){
        if (text[i+1] === '"'){ field += '"'; i++; }
        else { inQuotes = false; }
      } else { field += c; }
    } else {
      if (c === '"'){ inQuotes = true; }
      else if (c === ','){ row.push(field); field = ''; }
      else if (c === '\n'){ row.push(field); rows.push(row); row = []; field = ''; }
      else if (c === '\r'){ /* skip */ }
      else { field += c; }
    }
  }
  if (field.length || row.length){ row.push(field); rows.push(row); }
  return rows.filter(r => !(r.length === 1 && r[0] === ''));
}

  return rows.slice(1).map(r => {
    const obj = {};
    headers.forEach((h, idx) => { obj[h] = (r[idx] !== undefined ? r[idx] : '').trim(); });
    return obj;
  });
}

function tagTheme(details){
  const text = (details || '').toLowerCase();
  const scores = {};
  THEME_RULES.forEach(([theme, keywords]) => {
    keywords.forEach(kw => {
      if (text.indexOf(kw) !== -1) scores[theme] = (scores[theme] || 0) + 1;
    });
  });
  const entries = Object.entries(scores);
  if (entries.length === 0) return { primary: "General / Unclassified", possible: ["General / Unclassified"] };
  entries.sort((a, b) => b[1] - a[1]);
  return { primary: entries[0][0], possible: entries.map(e => e[0]) };
}

/* Mirrors grc_pipeline.py's build_graph_payload(), but running client-side
   in the browser directly against parsed CSV rows. */
function buildGraphPayloadFromRows(rows){
  const nodes = [], edges = [];
  const seenRisk = new Set();

  rows.forEach(row => {
    const issueId = (row.issue_id || '').trim();
    if (!issueId) return;
    const riskId = (row.risk_id || '').trim();
    const riskName = (row.risk_taxonomy || '').trim();
    const controlId = (row.control_id || '').trim();
    const hasGap = !controlId;

    if (riskId && !seenRisk.has(riskId)){
      nodes.push({ id: riskId, label: riskName || riskId, group: 'risk',
        details: { type: 'Risk Taxonomy', id: riskId, name: riskName || riskId } });
      seenRisk.add(riskId);
    }

    const { primary: theme, possible: possibleThemes } = tagTheme(row.issue_details);
    const probable = PROBABLE_CONTROL_BY_THEME[theme] || DEFAULT_PROBABLE_CONTROL;

    nodes.push({
      id: issueId,
      label: row.issue_title || issueId,
      group: hasGap ? 'issue_gap' : 'issue',
      details: {
        type: 'Issue', id: issueId, title: row.issue_title || issueId,
        description: row.issue_details || '', theme, possible_themes: possibleThemes,
        risk_taxonomy: riskName,
        control_status: hasGap ? 'GAP - No control mapped' : 'Mapped',
        probable_control: hasGap ? probable : null,
      },
    });
    if (riskId) edges.push({ from: riskId, to: issueId, label: 'categorized under' });

    if (hasGap){
      const gapId = 'GAP-' + issueId;
      nodes.push({ id: gapId, label: 'No Control Mapped', group: 'gap',
        details: { type: 'Control Gap', issue_id: issueId, theme, probable_control: probable } });
      edges.push({ from: issueId, to: gapId, label: 'control gap', dashed: true });
    } else {
      nodes.push({
        id: controlId, label: row.control_title || controlId, group: 'control',
        details: { type: 'Control', id: controlId, title: row.control_title || controlId,
          description: row.control_details || '', linked_issue: issueId,
          linked_risk_taxonomy: riskName, theme },
      });
      edges.push({ from: issueId, to: controlId, label: 'mitigated by' });
    }
  });

  return { nodes, edges };
}

function loadCSVText(text){
  const rows = rowsToObjects(parseCSV(text));
  const missing = REQUIRED_COLUMNS.filter(c => !(rows[0] && c in rows[0]));
  if (rows.length === 0 || missing.length){
    throw new Error(missing.length
      ? `Missing required column(s): ${missing.join(', ')}`
      : 'CSV appears to be empty.');
  }
  return buildGraphPayloadFromRows(rows);
}

function ensureCatalogNode(id){
  if (nodesDS.get(id)) return;
  const raw = GRAPH_DATA.nodes.find(n => n.id === id);
  if (raw) nodesDS.add(makeVisNode(raw));
}

function ensureEdge(fromId, toId, opts){
  const id = fromId + '__' + toId;
  if (edgesDS.get(id)) return;
  edgesDS.add(Object.assign({
    id, from: fromId, to: toId,
    color:{ color:'rgba(143,160,192,0.35)', highlight:'#4F9CF9' },
    width: 1,
    smooth:{ type:'continuous', roundness:0.4 },
  }, opts || {}));
}

function issueRiskEdge(issueId){ return GRAPH_DATA.edges.find(e => e.to === issueId && e.label === 'categorized under'); }
function issueControlEdge(issueId){ return GRAPH_DATA.edges.find(e => e.from === issueId && e.label === 'mitigated by'); }
function issueGapEdge(issueId){ return GRAPH_DATA.edges.find(e => e.from === issueId && e.label === 'control gap'); }

function issuesForTheme(themeName){
  return GRAPH_DATA.nodes.filter(n =>
    (n.group === 'issue' || n.group === 'issue_gap') &&
    n.details && (n.details.theme === themeName || (n.details.possible_themes || []).includes(themeName))
  );
}

function revealIssueRiskAndControl(issueId){
  const rEdge = issueRiskEdge(issueId);
  if (rEdge){ ensureCatalogNode(rEdge.from); ensureEdge(rEdge.from, rEdge.to, { color:{color:'rgba(167,139,250,0.4)', highlight:'#A78BFA'} }); }

  const cEdge = issueControlEdge(issueId);
  if (cEdge){
    ensureCatalogNode(cEdge.to);
    ensureEdge(cEdge.from, cEdge.to, { color:{color:'rgba(52,211,163,0.45)', highlight:'#34D3A3'} });
    return;
  }
  const gEdge = issueGapEdge(issueId);
  if (gEdge && !resolvedGaps.has(issueId)){
    ensureCatalogNode(gEdge.to);
    ensureEdge(gEdge.from, gEdge.to, { dashes:true, color:{color:'rgba(241,85,76,0.55)', highlight:'#F1554C'} });
  }
}

function toggleThemeForIssue(issueId, themeName, chipEl){
  const tId = themeNodeId(themeName);
  if (!selectedThemesByIssue[issueId]) selectedThemesByIssue[issueId] = new Set();
  const set = selectedThemesByIssue[issueId];
  const edgeId = issueId + '__' + tId;

  if (set.has(themeName)){
    set.delete(themeName);
    edgesDS.remove(edgeId);
    const stillUsed = edgesDS.get({ filter: e => e.from === tId || e.to === tId }).length > 0;
    if (!stillUsed) nodesDS.remove(tId);
    if (chipEl) chipEl.classList.remove('selected');
  } else {
    set.add(themeName);
    if (!nodesDS.get(tId)) nodesDS.add(makeThemeVisNode(themeName));
    ensureEdge(issueId, tId, { color:{color:'rgba(34,211,238,0.45)', highlight:'#22D3EE'} });
    if (chipEl) chipEl.classList.add('selected');
  }
}

function expandTheme(themeName){
  const tId = themeNodeId(themeName);
  const similar = issuesForTheme(themeName);
  similar.forEach(issueNode => {
    ensureCatalogNode(issueNode.id);
    ensureEdge(tId, issueNode.id, { color:{color:'rgba(34,211,238,0.35)', highlight:'#22D3EE'} });
    revealIssueRiskAndControl(issueNode.id);
  });
}

function computeRecommendedControls(issueId, themeSet){
  const recs = new Map(); // controlId -> {controlNode, viaIssues:[]}
  themeSet.forEach(themeName => {
    issuesForTheme(themeName).forEach(sim => {
      if (sim.id === issueId || sim.group !== 'issue') return;
      const cEdge = issueControlEdge(sim.id);
      if (!cEdge) return;
      const ctlNode = GRAPH_DATA.nodes.find(n => n.id === cEdge.to);
      if (!ctlNode) return;
      if (!recs.has(ctlNode.id)) recs.set(ctlNode.id, { controlNode: ctlNode, viaIssues: [] });
      recs.get(ctlNode.id).viaIssues.push(sim);
    });
  });
  return Array.from(recs.values());
}

function submitRecommendation(issueId, selectedControlIds){
  selectedControlIds.forEach(ctlId => {
    ensureCatalogNode(ctlId);
    ensureEdge(issueId, ctlId, { color:{color:'rgba(52,211,163,0.5)', highlight:'#34D3A3'} });
  });
  const gEdge = issueGapEdge(issueId);
  if (gEdge){ edgesDS.remove(issueId + '__' + gEdge.to); nodesDS.remove(gEdge.to); }

  // Now fully "in the network": also connect its own risk taxonomy, same as
  // any already-mapped issue gets by default.
  const rEdge = issueRiskEdge(issueId);
  if (rEdge){ ensureCatalogNode(rEdge.from); ensureEdge(rEdge.from, rEdge.to, { color:{color:'rgba(167,139,250,0.4)', highlight:'#A78BFA'} }); }

  nodesDS.update(Object.assign({ id: issueId }, STYLE.issue));

  resolvedGaps.add(issueId);
  dynamicResolvedCount += 1;
  refreshStats();
}

function renderRecommendations(issueId){
  const wrap = document.getElementById('recWrap');
  if (!wrap) return;
  const themeSet = selectedThemesByIssue[issueId] || new Set();

  if (themeSet.size === 0){
    wrap.innerHTML = `<div class="recommend-empty">Tag at least one theme above to see recommended controls drawn from similar issues.</div>`;
    return;
  }

  const recs = computeRecommendedControls(issueId, themeSet);
  if (recs.length === 0){
    wrap.innerHTML = `<div class="recommend-empty">No similar issue under the selected theme(s) has a mapped control yet.</div>`;
    return;
  }

  let html = `<div class="recommend-box"><div class="k">Recommended controls</div>`;
  recs.forEach(r => {
    const viaLabel = r.viaIssues.map(i => i.id).join(', ');
    html += `<label class="rec-item">
               <input type="checkbox" class="rec-cb" value="${r.controlNode.id}">
               <span class="txt">${r.controlNode.label}<span class="sub mono">${r.controlNode.id} &middot; via ${viaLabel}</span></span>
             </label>`;
  });
  html += `<button class="btn primary" id="submitRecBtn">Submit selected control(s)</button></div>`;
  wrap.innerHTML = html;

  document.getElementById('submitRecBtn').addEventListener('click', () => {
    const chosen = Array.from(wrap.querySelectorAll('.rec-cb:checked')).map(cb => cb.value);
    if (chosen.length === 0){ alert('Select at least one recommended control before submitting.'); return; }
    submitRecommendation(issueId, chosen);
    renderIssuePanel(nodesDS.get(issueId)._raw);
  });
}

function applyFilters(){
  const activeRisks  = [...document.querySelectorAll('.risk-cb:checked')].map(el => el.value);
  const activeThemes = [...document.querySelectorAll('.theme-cb:checked')].map(el => el.value);
  const gapsOnly = document.getElementById('gapsOnlyToggle').checked;
  const query = document.getElementById('searchInput').value.trim().toLowerCase();

  // Search can reach beyond whatever has been clicked into view so far -
  // pull in any matching issue/control that isn't in the graph yet.
  if (query){
    GRAPH_DATA.nodes.forEach(n => {
      if (n.group !== 'issue' && n.group !== 'issue_gap' && n.group !== 'control') return;
      if (nodesDS.get(n.id)) return;
      const hay = (n.label + ' ' + n.id + ' ' + ((n.details||{}).description||'')).toLowerCase();
      if (hay.includes(query)) nodesDS.add(makeVisNode(n));
    });
  }

  nodesDS.forEach(n => {
    const raw = n._raw;
    const d = raw.details || {};
    let visible = true;
    const rt = d.risk_taxonomy || d.name || d.linked_risk_taxonomy;
    if (raw.group !== 'risk' && raw.group !== 'theme'){
      if (rt && !activeRisks.includes(rt)) visible = false;
      if (d.theme && !activeThemes.includes(d.theme)) visible = false;
    }
    if (gapsOnly && !(raw.group === 'gap' || raw.group === 'issue_gap')) visible = false;
    if (query){
      const hay = (n.label + ' ' + n.id + ' ' + (d.description||'')).toLowerCase();
      if (!hay.includes(query)) visible = false;
    }
    if (raw.group === 'risk' || raw.group === 'theme') visible = true;
    nodesDS.update({ id: n.id, hidden: !visible });
  });

  edgesDS.forEach(e => {
    const fromNode = nodesDS.get(e.from), toNode = nodesDS.get(e.to);
    const bothVisible = fromNode && toNode && !fromNode.hidden && !toNode.hidden;
    edgesDS.update({ id: e.id, hidden: !bothVisible });
  });
}

document.getElementById('gapsOnlyToggle').addEventListener('change', applyFilters);

/* Pick a good starting issue for the guided view: a GAP issue (no control
   mapped) whose theme has the most other issues under it, so the very
   first click-through densifies the network noticeably rather than
   revealing just one or two extra nodes. Falls back to any issue if the
   dataset has no gaps at all. */
function pickBestEntryIssue(){
  const gapIssues = GRAPH_DATA.nodes.filter(n => n.group === 'issue_gap');
  const candidates = gapIssues.length ? gapIssues : GRAPH_DATA.nodes.filter(n => n.group === 'issue' || n.group === 'issue_gap');
  let best = candidates[0], bestCount = -1;
  candidates.forEach(c => {
    const theme = c.details.theme;
    const count = issuesForTheme(theme).length;
    if (count > bestCount){ bestCount = count; best = c; }
  });
  return best;
}

function clearGraphState(){
  nodesDS.clear();
  edgesDS.clear();
  Object.keys(selectedThemesByIssue).forEach(k => delete selectedThemesByIssue[k]);
  resolvedGaps.clear();
  dynamicResolvedCount = 0;
  refreshStats();
  panel.classList.remove('open');
}

function loadGuidedView(){
  clearGraphState();
  const entry = pickBestEntryIssue();
  if (entry) nodesDS.add(makeVisNode(entry));
  network.fit({ animation:true });
}

function loadFullView(){
  clearGraphState();
  GRAPH_DATA.nodes
    .filter(n => n.group === 'issue' || n.group === 'issue_gap')
    .forEach(n => nodesDS.add(makeVisNode(n)));
  GRAPH_DATA.nodes
    .filter(n => n.group === 'issue')
    .forEach(n => revealIssueRiskAndControl(n.id));
  network.fit({ animation:true });
}

const viewFullBtn   = document.getElementById('viewFullBtn');
const guidedHint = document.getElementById('guidedHint');

function setActiveViewUI(){
  viewGuidedBtn.classList.toggle('primary', currentView === 'guided');
  viewFullBtn.classList.toggle('primary', currentView === 'full');
  guidedHint.style.display = currentView === 'guided' ? '' : 'none';
  fullHint.style.display   = currentView === 'full' ? '' : 'none';
}

viewGuidedBtn.addEventListener('click', () => {
  currentView = 'guided';
  setActiveViewUI();
  loadGuidedView();
});
viewFullBtn.addEventListener('click', () => {
  currentView = 'full';
  setActiveViewUI();
  loadFullView();
});

document.getElementById('resetBtn').addEventListener('click', () => {
  document.querySelectorAll('.risk-cb, .theme-cb').forEach(el => el.checked = true);
  document.getElementById('gapsOnlyToggle').checked = false;
  document.getElementById('searchInput').value = '';
  if (currentView === 'guided') loadGuidedView(); else loadFullView();
});

let physicsOn = true;
document.getElementById('physicsBtn').addEventListener('click', (e) => {
  physicsOn = !physicsOn;
  network.setOptions({ physics:{ enabled: physicsOn } });
  e.target.textContent = physicsOn ? 'Freeze layout' : 'Resume layout';
});

const csvFileInput = document.getElementById('csvFileInput');
const dataStatusEl = document.getElementById('dataStatus');
document.getElementById('loadCsvBtn').addEventListener('click', () => csvFileInput.click());

function applyNewGraphData(newData, sourceLabel){
  GRAPH_DATA = newData;
  recomputeTotals();
  rebuildFilterCheckboxes();
  currentView = 'guided';
  setActiveViewUI();
  loadGuidedView();
  dataStatusEl.textContent = `Data: ${sourceLabel} (${totalIssues} issues, ${totalRisks} risk areas)`;
}

csvFileInput.addEventListener('change', (e) => {
  const file = e.target.files && e.target.files[0];
  if (!file) return;
  const reader = new FileReader();
  reader.onload = () => {
    try {
      const payload = loadCSVText(reader.result);
      if (payload.nodes.filter(n => n.group === 'issue' || n.group === 'issue_gap').length === 0){
        throw new Error('No issue rows found - check the CSV has data below the header row.');
      }
      applyNewGraphData(payload, file.name);
    } catch (err){
      alert('Could not load this CSV:\\n\\n' + err.message);
    }
    csvFileInput.value = ''; // allow re-selecting the same file later
  };
  reader.onerror = () => alert('Could not read that file.');
  reader.readAsText(file);
});

/* Default view on load */
setActiveViewUI();
loadGuidedView();
dataStatusEl.textContent = `Data: built-in demo (${totalIssues} issues, ${totalRisks} risk areas)`;

/* =========================================================================
   8. RESILIENT LIBRARY LOADER
   Tries a couple of different CDNs in turn (some networks/firewalls block
   one but not another). Only calls initApp() once vis-network has actually
   loaded; shows a clear message in place of a blank/broken page if every
   source is unreachable, instead of throwing "vis is not defined".
   ========================================================================= */
const VIS_SOURCES = [
  'https://cdnjs.cloudflare.com/ajax/libs/vis-network/10.0.2/standalone/umd/vis-network.min.js',
  'https://cdn.jsdelivr.net/npm/vis-network@10.0.2/standalone/umd/vis-network.min.js',
  'https://unpkg.com/vis-network@10.0.2/standalone/umd/vis-network.min.js',
];

function loadVis(sourceIndex){
  if (sourceIndex >= VIS_SOURCES.length){
    document.getElementById('graph').innerHTML = `
      <div class="cdn-error">
        <div class="box">
          <h3>Graph library couldn't load</h3>
          <p>This page loads its charting library (<code>vis-network</code>) from a public CDN at
          open time. All CDN sources were unreachable from this browser/network just now.</p>
          <p>Things to try: open this file with an active internet connection, check whether your
          network or firewall blocks <code>cdnjs.cloudflare.com</code>, <code>jsdelivr.net</code>, and
          <code>unpkg.com</code>, or try a different browser/network.</p>
        </div>
      </div>`;
    return;
  }
  const s = document.createElement('script');
  s.src = VIS_SOURCES[sourceIndex];
  s.onload = () => {
    if (typeof vis === 'undefined'){ loadVis(sourceIndex + 1); return; }
    try { initApp(); } catch(err){
      console.error('Graph init failed:', err);
      document.getElementById('graph').innerHTML =
        '<div class="cdn-error"><div class="box"><h3>Something went wrong rendering the graph</h3><p>' +
        (err && err.message ? err.message : 'Unknown error') + '</p></div></div>';
    }
  };
  s.onerror = () => loadVis(sourceIndex + 1);
  document.head.appendChild(s);
}

loadVis(0);
</script>
